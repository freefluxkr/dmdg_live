###########################
# 라이브러리 사용
import tensorflow as tf
import pandas as pd

###########################
# 1.과거의 data를 준비합니다.
file_path = 'https://raw.githubusercontent.com/blackdew/tensorflow1/master/csv/iris.csv'
iris = pd.read_csv(file_path)
iris.head()

# one_hot_encoding
iris = pd.get_dummies(iris)

# Y변수, X변수
X = iris[['꽃잎길이', '꽃잎폭', '꽃받침길이', '꽃받침폭']]
Y = iris[['품종_setosa', '품종_versicolor', '품종_virginica']]
print(X.shape, Y.shape)

###########################
# 2. 모델의 구조를 만듭니다
X = tf.keras.layers.Input(shape=[4])
Y = tf.keras.layers.Dense(3, activation='softmax')(X)
model = tf.keras.models.Model(X, Y)
model.compile(loss='categorical_crossentropy',
              metrics='accuracy')

###########################
# 3.data로 모델을 학습(FIT)합니다.
model.fit(X, Y, epochs=1000, verbose=0)
model.fit(X, Y, epochs=10)

###########################
# 4. 모델을 이용합니다
# 맨 처음 data 5개
print(model.predict(X[:5]))
print(Y[:5])

# 맨 마지막 data 5개
print(model.predict(X[-5:]))
print(Y[-5:])

###########################
# weights & bias 출력
print(model.get_weights())

###########################
# 라이브러리 사용
import tensorflow as tf
import pandas as pd

###########################
# 1.과거의 data를 준비합니다.
file_path = 'https://raw.githubusercontent.com/blackdew/tensorflow1/master/csv/iris.csv'
iris = pd.read_csv(file_path)

# one_hot_encoding
iris = pd.get_dummies(iris)

# Y변수, X변수
X = iris[['꽃잎길이', '꽃잎폭', '꽃받침길이', '꽃받침폭']]
Y = iris[['품종_setosa', '품종_versicolor', '품종_virginica']]
print(X.shape, Y.shape)

###########################
# 2. 모델의 구조를 만듭니다
X = tf.keras.layers.Input(shape=[4])
H = tf.keras.layers.Dense(8, activation="swish")(X)
H = tf.keras.layers.Dense(8, activation="swish")(H)
H = tf.keras.layers.Dense(8, activation="swish")(H)
Y = tf.keras.layers.Dense(3, activation='softmax')(H)
model = tf.keras.models.Model(X, Y)
model.compile(loss='categorical_crossentropy',
              metrics='accuracy')

# 모델 구조 확인
model.summary()

###########################
# 3.data로 모델을 학습(FIT)합니다.
model.fit(X, Y, epochs=100)

###########################
# 4. 모델을 이용합니다

소스코드

colab |  backend.ai

###########################
# 라이브러리 로딩
import glob
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import tensorflow as tf 

###########################
# 이미지 읽어서 data 준비하기
paths = glob.glob('./notMNIST_small/*/*.png')
paths = np.random.permutation(paths)
X = np.array([plt.imread(paths[i]) for i in range(len(paths))])
Y = np.array([paths[i].split('/')[-2] for i in range(len(paths))])
print(X.shape, Y.shape)

X = X.reshape(18724, 28, 28, 1)
Y = pd.get_dummies(Y)
print(X.shape, Y.shape)

###########################
# 모델을 완성합니다. 
X = tf.keras.layers.Input(shape=[28, 28, 1])

H = tf.keras.layers.Conv2D(6, kernel_size=5, padding='same', activation='swish')(X)
H = tf.keras.layers.MaxPool2D()(H)

H = tf.keras.layers.Conv2D(16, kernel_size=5, activation='swish')(H)
H = tf.keras.layers.MaxPool2D()(H)

H = tf.keras.layers.Flatten()(H)
H = tf.keras.layers.Dense(120, activation='swish')(H)
H = tf.keras.layers.Dense(84, activation='swish')(H)
Y = tf.keras.layers.Dense(10, activation='softmax')(H)

model = tf.keras.models.Model(X, Y)
model.compile(loss='categorical_crossentropy', metrics='accuracy')

###########################
# 모델을 학습
model.fit(X, Y, epochs=10)

###########################
# 모델을 이용합니다. 
pred = model.predict(X[0:5])
pd.DataFrame(pred).round(2)

# 정답 확인
Y[0:5]

# 모델 확인
model.summary()
