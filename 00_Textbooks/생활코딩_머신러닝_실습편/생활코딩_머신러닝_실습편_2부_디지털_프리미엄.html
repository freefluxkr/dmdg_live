###########################
# 라이브러리 사용
import tensorflow as tf
import pandas as pd

###########################
# with reshape

# data를 준비하고
(X, Y), _ = tf.keras.datasets.mnist.load_data()
X = X.reshape(60000, 784)
Y = pd.get_dummies(Y)
print(X.shape, Y.shape)

# 모델을 만들고
X = tf.keras.layers.Input(shape=[784])
H = tf.keras.layers.Dense(84, activation='swish')(X)
Y = tf.keras.layers.Dense(10, activation='softmax')(H)
model = tf.keras.models.Model(X, Y)
model.compile(loss='categorical_crossentropy', metrics='accuracy')

# 모델을 학습하고
model.fit(X, Y, epochs=10)

# 모델을 이용합니다. 
pred = model.predict(X[0:5])
print(pd.DataFrame(pred).round(2))
print(Y[0:5])

# with flatten

###########################
# 라이브러리 사용
import tensorflow as tf
import pandas as pd

###########################
# data를 준비하고
(X, Y), _ = tf.keras.datasets.mnist.load_data()
X = X.reshape(60000, 28, 28, 1)
Y = pd.get_dummies(Y)
print(X.shape, Y.shape)

###########################
# 모델을 만들고
X = tf.keras.layers.Input(shape=[28, 28, 1])
H = tf.keras.layers.Conv2D(3, kernel_size=5, activation='swish')(X)
H = tf.keras.layers.Conv2D(6, kernel_size=5, activation='swish')(H)
H = tf.keras.layers.Flatten()(H)
H = tf.keras.layers.Dense(84, activation='swish')(H)
Y = tf.keras.layers.Dense(10, activation='softmax')(H)
model = tf.keras.models.Model(X, Y)
model.compile(loss='categorical_crossentropy', metrics='accuracy')

###########################
# 모델을 학습하고
model.fit(X, Y, epochs=10)

###########################
# 모델을 이용합니다. 
pred = model.predict(X[0:5])
pd.DataFrame(pred).round(2)

# 정답 확인
Y[0:5]

# 모델 확인
model.summary()

###########################
# 라이브러리 사용
import tensorflow as tf
import pandas as pd

###########################
# data를 준비합니다. 
(X, Y), _ = tf.keras.datasets.cifar10.load_data()
Y = pd.get_dummies(Y.reshape(50000))
print(X.shape, Y.shape)

###########################
# 모델을 완성합니다. 
X = tf.keras.layers.Input(shape=[32, 32, 3])

H = tf.keras.layers.Conv2D(6, kernel_size=5, activation='swish')(X)

H = tf.keras.layers.Conv2D(16, kernel_size=5, activation='swish')(H)
H = tf.keras.layers.MaxPool2D()(H)

###########################

model.fit(X, Y, epochs=10)

###########################

소스코드

###########################
# 라이브러리 로딩
import glob
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import tensorflow as tf 

###########################
# 이미지 읽어서 data 준비하기
paths = glob.glob('./notMNIST_small/*/*.png')
paths = np.random.permutation(paths)
X = np.array([plt.imread(paths[i]) for i in range(len(paths))])
Y = np.array([paths[i].split('/')[-2] for i in range(len(paths))])
print(X.shape, Y.shape)

X = X.reshape(18724, 28, 28, 1)
Y = pd.get_dummies(Y)
print(X.shape, Y.shape)

###########################
# 모델을 완성합니다. 
X = tf.keras.layers.Input(shape=[28, 28, 1])

H = tf.keras.layers.Conv2D(6, kernel_size=5, padding='same', activation='swish')(X)
H = tf.keras.layers.MaxPool2D()(H)

H = tf.keras.layers.Conv2D(16, kernel_size=5, activation='swish')(H)
H = tf.keras.layers.MaxPool2D()(H)

H = tf.keras.layers.Flatten()(H)
H = tf.keras.layers.Dense(120, activation='swish')(H)
H = tf.keras.layers.Dense(84, activation='swish')(H)
Y = tf.keras.layers.Dense(10, activation='softmax')(H)

model = tf.keras.models.Model(X, Y)
model.compile(loss='categorical_crossentropy', metrics='accuracy')

###########################
# 모델을 학습
model.fit(X, Y, epochs=10)

###########################
# 모델을 이용합니다. 
pred = model.predict(X[0:5])
pd.DataFrame(pred).round(2)

# 정답 확인
Y[0:5]

# 모델 확인
model.summary()
