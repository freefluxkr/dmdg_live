class WeightManager:
    def __init__(self):
        # 기본 가중치 설정
        self.weights = {
            'Area_Size': 0.15,
            'House_Age': -0.10,
            'Transport_Score': 0.20,
            'Safety_Score': 0.25,
            'Noise_Level': -0.15,
            'Age_Impact': 0.05,
        }

    def update_weight(self, key, value):
        """사용자 입력 값에 따라 가중치를 수정합니다."""
        if key in self.weights:
            self.weights[key] = value
            print(f"가중치 업데이트 완료: {key} -> {value}")
        else:
            print(f"오류: 존재하지 않는 가중치 키입니다 ({key})")

    def get_weights(self):
        """현재 적용된 가중치를 반환합니다."""
        return self.weights
