import joblib
import pandas as pd
import os

class HousingRecommender:
    def __init__(self, model_path="housing_model.pkl"):
        self.model_path = model_path
        self.model = None
        self.load_model()

    def load_model(self):
        """저장된 머신러닝 모델을 로드합니다."""
        if os.path.exists(self.model_path):
            self.model = joblib.load(self.model_path)
            print("저장된 추천 모델을 성공적으로 불러왔습니다.")
        else:
            print("오류: 저장된 모델을 찾을 수 없습니다. 먼저 학습을 진행해주세요.")

    def recommend(self, user_profile):
        """
        신규 사용자 정보를 받아 최적의 주거 형태를 추천합니다.
        user_profile: Feature들을 담고 있는 딕셔너리
        """
        if self.model is None:
            return "모델이 로드되지 않아 추천을 진행할 수 없습니다."

        # DataFrame으로 변환 (단일 샘플 예측)
        df = pd.DataFrame([user_profile])
        prediction = self.model.predict(df)
        return prediction[0]
