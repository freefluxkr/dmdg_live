from weight_manager import WeightManager
from model_trainer import ModelTrainer
from recommender import HousingRecommender
import os

def main():
    print("="*50)
    print(" 주거 추천 시스템 파이프라인 (독립 모듈화 버전)")
    print("="*50)

    # 1. 가중치 관리 모듈 초기화 및 사용
    print("\n[1단계: 사용자 입력 기반 가중치 수정]")
    wm = WeightManager()
    print("기존 가중치:", wm.get_weights())
    # 사용자 입력에 따른 가중치 수정 (예시)
    wm.update_weight('Transport_Score', 0.35)
    wm.update_weight('Safety_Score', 0.40)
    print("수정된 가중치:", wm.get_weights())

    # 2. 모델 학습 모듈
    print("\n[2단계: 모델 학습 및 로컬 저장]")
    data_path = "synthetic_survey_data_custom.csv"
    model_path = "housing_model.pkl"
    
    trainer = ModelTrainer(data_path=data_path, model_save_path=model_path)
    if os.path.exists(data_path):
        trainer.train_model()
    else:
        print(f"데이터 파일이 없습니다. 기존의 generate_advanced_survey.py를 통해 {data_path} 를 먼저 생성해주세요.")
        return

    # 3. 추천 시스템 추론 모듈
    print("\n[3단계: 신규 사용자 기반 주거 추천 추론]")
    recommender = HousingRecommender(model_path=model_path)
    
    # 추천을 위한 임의의 신규 사용자 프로필 (입력 폼 등으로 받았다고 가정)
    new_user = {
        'Area_Size': 55.0,
        'House_Age': 5,
        'Transport_Score': 90,
        'Safety_Score': 40,
        'Noise_Level': 80,
        'Age': 28,
        'Gender': 'Female',
        'Job': 'Office',
        'Building_Type': 'Officetel'
    }
    
    print("\n[입력된 신규 사용자 프로필]")
    for k, v in new_user.items():
        print(f"- {k}: {v}")
        
    result = recommender.recommend(new_user)
    print(f"\n>> 🎯 AI 추천 결과: {result}")

if __name__ == "__main__":
    main()
