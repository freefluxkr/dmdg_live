pub const CENTER_X: f32 = 320.0;
pub const MAX_SPAN: f32 = 320.0;

#[inline]
pub fn calculate_dxball_sound_pan(pos_x: f32) -> f32 {
    if pos_x.is_nan() {
        return 0.0;
    }
    let offset = pos_x - CENTER_X;
    let normalized = offset / MAX_SPAN;
    normalized.clamp(-1.0, 1.0)
}
