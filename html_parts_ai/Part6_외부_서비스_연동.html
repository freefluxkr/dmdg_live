16.1 Function Calling / Tool Use란 무엇인가

Function Calling(함수 호출)은 LLM이 응답을 생성하는 과정에서 미리 정의된 외부 함수를 호출하고, 그 결과를 받아 최종 답변에 반영할 수 있도록 하는 기능입니다. OpenAI에서는 Function Calling, Google Gemini와 Anthropic Claude에서는 Tool Use라는 이름으로 제공하지만 동작 원리는 동일합니다.

LLM은 스스로 코드를 실행하거나 인터넷을 검색하지 않습니다. Function Calling은 LLM이 "어떤 함수를 어떤 인자로 호출해야 하는지"를 결정하면, 실제 실행은 개발자 코드가 담당하는 방식입니다.

동작 흐름

Function Calling의 전체 흐름은 다음과 같습니다.

💡 핵심: LLM은 함수를 직접 실행하지 않습니다. "어떤 함수를 어떤 인자로 호출할지"만 결정하고, 실제 실행은 항상 앱(개발자 코드) 쪽에서 책임집니다.

16.2 OpenAI Function Calling 구현

OpenAI의 Function Calling은 tools 파라미터에 JSON Schema 형식으로 함수를 정의하고, 모델이 필요 시 해당 함수 호출을 요청하는 방식으로 동작합니다.

함수 정의 (JSON Schema)

import json
import openai

client = openai.OpenAI()

# 함수 스키마 정의
tools = [
    {
        "type": "function",
        "function": {
            "name": "search_web",
            "description": "웹에서 최신 정보를 검색합니다. 실시간 뉴스, 날씨, 최신 사건 등을 조회할 때 사용합니다.",
            "parameters": {
                "type": "object",
                "properties": {
                    "query": {
                        "type": "string",
                        "description": "검색할 키워드 또는 문장"
                    },
                    "num_results": {
                        "type": "integer",
                        "description": "반환할 검색 결과 수 (기본값: 5)",
                        "default": 5
                    }
                },
                "required": ["query"]
            }
        }
    }
]

1단계: 모델에 함수 목록 전달 및 호출 요청 수신

def run_function_calling(user_message: str):
    messages = [{"role": "user", "content": user_message}]

    # 1단계: 모델에 tools 전달하여 함수 호출 여부 판단 요청
    response = client.chat.completions.create(
        model="gpt-4o",
        messages=messages,
        tools=tools,
        tool_choice="auto"  # "auto": 모델이 자동 판단, "required": 반드시 호출
    )

    response_message = response.choices[0].message

    # 함수 호출이 필요 없는 경우 바로 응답 반환
    if not response_message.tool_calls:
        return response_message.content

    return response_message

2단계: 함수 실행 및 결과 반환

def execute_tool_calls(response_message, messages: list):
    """모델이 요청한 함수를 실제로 실행합니다."""
    messages.append(response_message)  # 어시스턴트 메시지 추가

    for tool_call in response_message.tool_calls:
        function_name = tool_call.function.name
        function_args = json.loads(tool_call.function.arguments)

        print(f"[함수 호출] {function_name}({function_args})")

        # 실제 함수 실행
        if function_name == "search_web":
            result = search_web(**function_args)
        else:
            result = {"error": f"알 수 없는 함수: {function_name}"}

        # 함수 실행 결과를 tool 메시지로 추가
        messages.append({
            "role": "tool",
            "tool_call_id": tool_call.id,
            "content": json.dumps(result, ensure_ascii=False)
        })

    return messages

3단계: 최종 응답 생성

def get_final_response(messages: list) -> str:
    """함수 결과를 포함한 최종 응답을 생성합니다."""
    final_response = client.chat.completions.create(
        model="gpt-4o",
        messages=messages
    )
    return final_response.choices[0].message.content


# 전체 흐름 통합
def chat_with_tools(user_message: str) -> str:
    messages = [{"role": "user", "content": user_message}]

    # 1단계: 함수 호출 요청 수신
    response = client.chat.completions.create(
        model="gpt-4o",
        messages=messages,
        tools=tools,
        tool_choice="auto"
    )
    response_message = response.choices[0].message

    if not response_message.tool_calls:
        return response_message.content

    # 2단계: 함수 실행
    messages = execute_tool_calls(response_message, messages)

    # 3단계: 최종 응답 생성
    return get_final_response(messages)


# 테스트
result = chat_with_tools("오늘 서울 날씨 알려줘")
print(result)

16.3 Gemini의 Function Calling 구현 방법

Google Gemini에서는 tools 파라미터에 FunctionDeclaration을 사용하여 함수를 정의합니다. google-generativeai 라이브러리를 사용합니다.

import google.generativeai as genai
import json

genai.configure(api_key="YOUR_GEMINI_API_KEY")

# 함수 선언 정의
search_web_declaration = genai.protos.FunctionDeclaration(
    name="search_web",
    description="웹에서 최신 정보를 검색합니다.",
    parameters=genai.protos.Schema(
        type=genai.protos.Type.OBJECT,
        properties={
            "query": genai.protos.Schema(
                type=genai.protos.Type.STRING,
                description="검색할 키워드 또는 문장"
            ),
            "num_results": genai.protos.Schema(
                type=genai.protos.Type.INTEGER,
                description="반환할 검색 결과 수"
            )
        },
        required=["query"]
    )
)

# Tool 객체 생성
search_tool = genai.protos.Tool(
    function_declarations=[search_web_declaration]
)

def chat_with_gemini_tools(user_message: str) -> str:
    model = genai.GenerativeModel(
        model_name="gemini-1.5-pro",
        tools=[search_tool]
    )
    chat = model.start_chat()

    # 1단계: 사용자 메시지 전송
    response = chat.send_message(user_message)
    candidate = response.candidates[0]

    # 함수 호출 요청 확인
    if candidate.content.parts[0].function_call.name:
        function_call = candidate.content.parts[0].function_call
        function_name = function_call.name
        function_args = dict(function_call.args)

        print(f"[Gemini 함수 호출] {function_name}({function_args})")

        # 2단계: 함수 실행
        if function_name == "search_web":
            result = search_web(**function_args)

        # 3단계: 함수 결과 반환 및 최종 응답 생성
        function_response = genai.protos.Part(
            function_response=genai.protos.FunctionResponse(
                name=function_name,
                response={"result": result}
            )
        )
        final_response = chat.send_message(function_response)
        return final_response.text
    else:
        return candidate.content.parts[0].text


result = chat_with_gemini_tools("최신 AI 뉴스를 알려줘")
print(result)

Anthropic Claude에서는 Tool Use라는 이름으로 동일한 기능을 제공합니다. anthropic 라이브러리를 사용하며, tools 파라미터에 input_schema 형식으로 도구를 정의합니다.

import anthropic
import json

client = anthropic.Anthropic(api_key="YOUR_CLAUDE_API_KEY")

# 도구 정의
tools = [
    {
        "name": "search_web",
        "description": "웹에서 최신 정보를 검색합니다. 실시간 뉴스, 날씨, 최신 사건 등을 조회할 때 사용합니다.",
        "input_schema": {
            "type": "object",
            "properties": {
                "query": {
                    "type": "string",
                    "description": "검색할 키워드 또는 문장"
                },
                "num_results": {
                    "type": "integer",
                    "description": "반환할 검색 결과 수 (기본값: 5)"
                }
            },
            "required": ["query"]
        }
    }
]

    messages = [{"role": "user", "content": user_message}]

    # 1단계: 도구 사용 요청
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        tools=tools,
        messages=messages
    )

    # 도구 호출 여부 확인 (stop_reason이 "tool_use"인 경우)
    if response.stop_reason == "tool_use":
        # 어시스턴트 응답을 메시지에 추가
        messages.append({"role": "assistant", "content": response.content})

        tool_results = []
        for content_block in response.content:
            if content_block.type == "tool_use":
                tool_name = content_block.name
                tool_input = content_block.input
                tool_use_id = content_block.id

                print(f"[Claude 도구 호출] {tool_name}({tool_input})")

                # 2단계: 도구 실행
                if tool_name == "search_web":
                    result = search_web(**tool_input)

                tool_results.append({
                    "type": "tool_result",
                    "tool_use_id": tool_use_id,
                    "content": json.dumps(result, ensure_ascii=False)
                })

        # 3단계: 도구 결과를 포함한 최종 응답 생성
        messages.append({"role": "user", "content": tool_results})

        final_response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
        return final_response.content[0].text
    else:
        return response.content[0].text


result = chat_with_claude_tools("현재 비트코인 가격을 알려줘")
print(result)

16.5 Google Search API 연동

Google Custom Search JSON API를 사용하면 프로그래밍 방식으로 Google 검색 결과를 가져올 수 있습니다. Function Calling과 결합하여 LLM에 실시간 정보를 제공하는 데 활용합니다.

API 키 발급 절차

  1. Google Cloud Console에서 프로젝트 생성
  2. Custom Search API 활성화
  3. API 키 발급 (Credentials 메뉴)
  4. Programmable Search Engine에서 검색 엔진 생성
  5. Search Engine ID (cx) 확인

검색 함수 구현

import requests
import os

GOOGLE_API_KEY = os.environ.get("GOOGLE_API_KEY")
GOOGLE_CSE_ID = os.environ.get("GOOGLE_CSE_ID")  # Custom Search Engine ID

def search_web(query: str, num_results: int = 5) -> dict:
    """
    Google Custom Search API를 사용하여 웹을 검색합니다.

    Args:
        query: 검색 키워드
        num_results: 반환할 결과 수 (최대 10)

    Returns:
        검색 결과 딕셔너리
    """
    url = "https://www.googleapis.com/customsearch/v1"
    params = {
        "key": GOOGLE_API_KEY,
        "cx": GOOGLE_CSE_ID,
        "q": query,
        "num": min(num_results, 10),  # API 제한: 최대 10개
        "hl": "ko"  # 한국어 결과 우선
    }

    try:
        response = requests.get(url, params=params, timeout=10)
        response.raise_for_status()
        data = response.json()

        results = []
        for item in data.get("items", []):
            results.append({
                "title": item.get("title", ""),
                "link": item.get("link", ""),
                "snippet": item.get("snippet", ""),
                "displayLink": item.get("displayLink", "")
            })

        return {
            "query": query,
            "total_results": data.get("searchInformation", {}).get("totalResults", "0"),
            "results": results
        }

    except requests.exceptions.RequestException as e:
        return {"error": str(e), "query": query, "results": []}


# 테스트
result = search_web("2024년 노벨 물리학상 수상자", num_results=3)
for item in result["results"]:
    print(f"제목: {item['title']}")
    print(f"URL: {item['link']}")
    print(f"요약: {item['snippet']}")
    print("---")

16.6 웹 검색 결과를 LLM에 전달하여 응답 생성

검색 결과를 LLM에 효과적으로 전달하려면 검색 결과를 적절히 포맷팅하여 컨텍스트로 제공해야 합니다.

검색 결과 포맷팅

def format_search_results(search_result: dict) -> str:
    """검색 결과를 LLM이 이해하기 쉬운 형식으로 변환합니다."""
    if "error" in search_result:
        return f"검색 중 오류가 발생했습니다: {search_result['error']}"

    query = search_result.get("query", "")
    results = search_result.get("results", [])

    if not results:
        return f"'{query}'에 대한 검색 결과가 없습니다."

    formatted = f"검색어: {query}\n검색 결과:\n\n"
    for i, item in enumerate(results, 1):
        formatted += f"{i}. **{item['title']}**\n"
        formatted += f"   출처: {item['displayLink']}\n"
        formatted += f"   내용: {item['snippet']}\n\n"

    return formatted

검색 결과 기반 응답 생성 (OpenAI)

def search_and_answer(user_question: str) -> str:
    """
    사용자 질문에 대해 웹 검색 후 LLM으로 답변을 생성합니다.
    """
    # 1단계: 검색 수행
    search_result = search_web(user_question, num_results=5)
    formatted_results = format_search_results(search_result)

    # 2단계: 검색 결과를 컨텍스트로 LLM에 전달
    system_prompt = """당신은 웹 검색 결과를 바탕으로 정확한 정보를 제공하는 AI 어시스턴트입니다.
제공된 검색 결과를 참고하여 사용자의 질문에 답변하되, 다음 사항을 지켜주세요:
1. 검색 결과에서 관련 정보를 찾아 정확히 인용합니다.
2. 불확실한 정보는 "검색 결과에 따르면" 등의 표현으로 출처를 명시합니다.
3. 검색 결과에 없는 내용은 추측하지 않습니다."""

    messages = [
        {"role": "system", "content": system_prompt},
        {"role": "user", "content": f"다음 검색 결과를 바탕으로 질문에 답해주세요.\n\n{formatted_results}\n\n질문: {user_question}"}
    ]

    response = client.chat.completions.create(
        model="gpt-4o",
        messages=messages,
        temperature=0.3  # 정확성을 위해 낮은 temperature 사용
    )

    return response.choices[0].message.content


# 테스트
answer = search_and_answer("최근 AI 규제 법안 현황은 어떻게 되나요?")
print(answer)

16.7 실시간 정보 기반 대화 시스템

                "role": "system",
                "content": """당신은 실시간 웹 검색 기능을 갖춘 AI 어시스턴트입니다.
최신 정보가 필요한 질문에는 search_web 도구를 사용하여 정확한 정보를 제공합니다.
검색 결과를 바탕으로 명확하고 유익한 답변을 한국어로 제공합니다."""
            })

    def _process_tool_calls(self, response_message) -> list:
        """도구 호출을 처리하고 결과를 반환합니다."""
        self.messages.append(response_message)
        tool_results = []

        for tool_call in response_message.tool_calls:
            function_name = tool_call.function.name
            function_args = json.loads(tool_call.function.arguments)

            print(f"  [검색 중] {function_args.get('query', '')}")

            if function_name == "search_web":
                result = search_web(**function_args)
                formatted = format_search_results(result)
            else:
                formatted = json.dumps({"error": f"알 수 없는 도구: {function_name}"})

            tool_results.append({
                "role": "tool",
                "tool_call_id": tool_call.id,
                "content": formatted
            })

        self.messages.extend(tool_results)
        return tool_results

    def chat(self, user_message: str) -> str:
        """사용자 메시지를 처리하고 응답을 반환합니다."""
        self.messages.append({"role": "user", "content": user_message})

        # 최대 3번의 도구 호출 루프 (무한 루프 방지)
        for _ in range(3):
            response = client.chat.completions.create(
                model="gpt-4o",
                messages=self.messages,
                tools=tools,
                tool_choice="auto"
            )
            response_message = response.choices[0].message

            if not response_message.tool_calls:
                # 도구 호출 없음: 최종 응답 반환
                self.messages.append({
                    "role": "assistant",
                    "content": response_message.content
                })
                return response_message.content

            # 도구 호출 처리
            self._process_tool_calls(response_message)

        return "응답 생성 중 오류가 발생했습니다."

    def reset(self):
        """대화 기록을 초기화합니다."""
        system_msg = self.messages[0] if self.messages else None
        self.messages = [system_msg] if system_msg else []

# ── 검색 함수 ──────────────────────────────────────────────────────────────────

def search_web(query: str, num_results: int = 5) -> dict:
    """Google Custom Search API로 웹 검색을 수행합니다."""
    url = "https://www.googleapis.com/customsearch/v1"
    params = {
        "key": GOOGLE_API_KEY,
        "cx": GOOGLE_CSE_ID,
        "q": query,
        "num": min(num_results, 10),
        "hl": "ko"
    }
    try:
        resp = requests.get(url, params=params, timeout=10)
        resp.raise_for_status()
        data = resp.json()
        results = [
            {
                "title": item.get("title", ""),
                "link": item.get("link", ""),
                "snippet": item.get("snippet", ""),
                "source": item.get("displayLink", "")
            }
            for item in data.get("items", [])
        ]
        return {"query": query, "results": results}
    except Exception as e:
        return {"query": query, "results": [], "error": str(e)}


def format_search_results(result: dict) -> str:
    """검색 결과를 읽기 좋은 형식으로 변환합니다."""
    if not result.get("results"):
        return f"'{result['query']}'에 대한 결과가 없습니다."

    lines = [f"[검색어: {result['query']}]\n"]
    for i, item in enumerate(result["results"], 1):
        lines.append(f"{i}. {item['title']}")
        lines.append(f"   출처: {item['source']}")
        lines.append(f"   {item['snippet']}\n")
    return "\n".join(lines)

[행동 지침]
- 최신 정보나 실시간 데이터가 필요한 경우 반드시 search_web 도구를 사용합니다.
- 일반 지식이나 개념 설명은 도구 없이 직접 답변합니다.
- 검색 결과를 인용할 때는 출처를 함께 언급합니다.
- 한국어로 친절하고 명확하게 답변합니다."""


# ── GPT 기반 검색 봇 ───────────────────────────────────────────────────────────

class GPTSearchBot:
    def __init__(self):
        self.messages = [{"role": "system", "content": SYSTEM_PROMPT}]

    def _execute_tools(self, tool_calls) -> list:
        tool_messages = []
        for tc in tool_calls:
            args = json.loads(tc.function.arguments)
            print(f"  🔍 검색: {args.get('query', '')}")
            result = search_web(**args)
            tool_messages.append({
                "role": "tool",
                "tool_call_id": tc.id,
                "content": format_search_results(result)
            })
        return tool_messages

    def chat(self, user_message: str) -> str:
        self.messages.append({"role": "user", "content": user_message})

        for _ in range(3):  # 최대 3회 도구 호출 루프
            resp = openai_client.chat.completions.create(
                model="gpt-4o",
                messages=self.messages,
                tools=TOOLS,
                tool_choice="auto"
            )
            msg = resp.choices[0].message

            if not msg.tool_calls:
                self.messages.append({"role": "assistant", "content": msg.content})
                return msg.content

            # 도구 호출 처리
            self.messages.append(msg)
            tool_results = self._execute_tools(msg.tool_calls)
            self.messages.extend(tool_results)

        return "오류: 응답 생성에 실패했습니다."

search_declaration = genai.protos.FunctionDeclaration(
    name="search_web",
    description="웹에서 최신 정보를 검색합니다.",
    parameters=genai.protos.Schema(
        type=genai.protos.Type.OBJECT,
        properties={
            "query": genai.protos.Schema(
                type=genai.protos.Type.STRING,
                description="검색할 키워드나 문장"
            ),
            "num_results": genai.protos.Schema(
                type=genai.protos.Type.INTEGER,
                description="반환할 결과 수"
            )
        },
        required=["query"]
    )
)
gemini_tool = genai.protos.Tool(function_declarations=[search_declaration])


class GeminiSearchBot:
    def __init__(self):
        self.model = genai.GenerativeModel(
            model_name="gemini-1.5-pro",
            tools=[gemini_tool],
            system_instruction=SYSTEM_PROMPT
        )
        self.chat = self.model.start_chat()

    def chat_message(self, user_message: str) -> str:
        response = self.chat.send_message(user_message)

        for _ in range(3):
            part = response.candidates[0].content.parts[0]
            if not hasattr(part, "function_call") or not part.function_call.name:
                return part.text

            fc = part.function_call
            args = dict(fc.args)
            print(f"  🔍 검색 (Gemini): {args.get('query', '')}")
            result = search_web(**args)

            function_response = genai.protos.Part(
                function_response=genai.protos.FunctionResponse(
                    name=fc.name,
                    response={"result": format_search_results(result)}
                )
            )
            response = self.chat.send_message(function_response)

        return response.text

플랫폼별 전환 포인트 요약

항목 GPT (OpenAI) Gemini Claude
클라이언트 openai.OpenAI() genai.GenerativeModel() anthropic.Anthropic()
도구 정의 tools 리스트 + JSON Schema FunctionDeclaration tools 리스트 + input_schema
호출 판단 msg.tool_calls 존재 여부 part.function_call.name stop_reason == "tool_use"
결과 전달 role: "tool" 메시지 FunctionResponse Part type: "tool_result"
멀티턴 messages 리스트 ChatSession 객체 messages 리스트

핵심 정리

  • Function Calling / Tool Use는 LLM이 외부 함수 실행을 요청하고, 개발자 코드가 실행 후 결과를 LLM에 반환하는 패턴입니다.
  • OpenAI는 tool_calls / role: "tool", Gemini는 FunctionDeclaration / FunctionResponse, Claude는 input_schema / tool_result 방식으로 구현합니다.
  • Google Custom Search API와 결합하면 LLM의 지식 한계를 넘어 실시간 정보를 응답에 반영할 수 있습니다.
  • 실시간 대화 시스템에서는 멀티턴 메시지 관리와 도구 호출 루프를 올바르게 구현하는 것이 핵심입니다.

이 챕터에서는 실제 세계의 데이터를 LLM과 연결하는 핵심 기술인 외부 API 연동을 다룹니다. 날씨, 뉴스, YouTube 등 다양한 공개 API를 통해 실시간 데이터를 수집하고, 이를 자연어로 가공하여 사용자에게 의미 있는 정보를 제공하는 방법을 학습합니다.

import os
import requests
from dotenv import load_dotenv

load_dotenv()

OPENWEATHER_API_KEY = os.getenv("OPENWEATHER_API_KEY")
BASE_URL = "https://api.openweathermap.org/data/2.5"

def get_current_weather(city: str, lang: str = "kr") -> dict:
    """도시명으로 현재 날씨를 조회합니다."""
    url = f"{BASE_URL}/weather"
    params = {
        "q": city,
        "appid": OPENWEATHER_API_KEY,
        "units": "metric",   # 섭씨 온도
        "lang": lang         # 한국어 설명
    }
    response = requests.get(url, params=params)
    response.raise_for_status()
    return response.json()

def get_forecast(city: str, days: int = 5) -> dict:
    """5일간 3시간 단위 예보를 조회합니다."""
    url = f"{BASE_URL}/forecast"
    params = {
        "q": city,
        "appid": OPENWEATHER_API_KEY,
        "units": "metric",
        "lang": "kr",
        "cnt": days * 8   # 하루 8개 시간대 (3시간 단위)
    }
    response = requests.get(url, params=params)
    response.raise_for_status()
    return response.json()

def search_news(
    keyword: str,
    language: str = "ko",
    days_ago: int = 3,
    max_articles: int = 10
) -> list[dict]:
    """키워드로 최근 뉴스를 검색합니다."""
    from_date = (datetime.now() - timedelta(days=days_ago)).strftime("%Y-%m-%d")

    url = f"{NEWS_BASE_URL}/everything"
    params = {
        "q": keyword,
        "language": language,
        "from": from_date,
        "sortBy": "relevancy",  # relevancy, popularity, publishedAt
        "pageSize": max_articles,
        "apiKey": NEWS_API_KEY
    }
    response = requests.get(url, params=params)
    response.raise_for_status()

    data = response.json()
    return data.get("articles", [])

def get_top_headlines(

def summarize_single_article(article: dict) -> str:

    description = article.get("description", "")
    content = article.get("content", "")

    # content는 NewsAPI 무료 플랜에서 200자로 잘릴 수 있음

            },
            {
                "role": "user",
                "content": f"제목: {title}\n\n내용: {full_text}"
            }
        ],
        max_tokens=200
    )
    return response.choices[0].message.content

def create_news_digest(keyword: str, max_articles: int = 5) -> str:
    """키워드 관련 뉴스를 수집하고 종합 다이제스트를 생성합니다."""
    articles = search_news(keyword, max_articles=max_articles)

    if not articles:
        return f"'{keyword}' 관련 최근 뉴스를 찾을 수 없습니다."

    # 각 기사 요약 수집
    summaries = []
    for i, article in enumerate(articles, 1):
        title = article.get("title", "제목 없음")
        source = article.get("source", {}).get("name", "출처 불명")
        summary = summarize_single_article(article)
        summaries.append(f"{i}. [{source}] {title}\n   → {summary}")

    summaries_text = "\n\n".join(summaries)

    # 종합 트렌드 분석
    response = client.chat.completions.create(
        model="gpt-4o-mini",
        messages=[
            {
                "role": "system",
                "content": (
                    "뉴스 요약들을 분석하여 전체적인 트렌드와 핵심 인사이트를 "
                    "3~4문장으로 종합해 주세요."
                )
            },
            {
                "role": "user",
                "content": (
                    f"'{keyword}' 관련 최신 뉴스 요약:\n\n{summaries_text}\n\n"
                    "위 뉴스들의 전반적인 트렌드를 분석해 주세요."
                )
            }
        ]
    )
    trend_analysis = response.choices[0].message.content

    result = f"## '{keyword}' 뉴스 다이제스트\n\n"
    result += f"### 개별 기사 요약\n{summaries_text}\n\n"
    result += f"### 트렌드 분석\n{trend_analysis}"
    return result

17.4 API 응답 데이터 가공과 자연어 변환 패턴

외부 API로부터 받은 JSON 구조체를 LLM이 효과적으로 처리할 수 있는 텍스트로 변환하는 것은 중요한 엔지니어링 기술입니다. 잘못된 데이터 가공은 LLM의 답변 품질을 저하시킬 수 있습니다.

데이터 가공 3원칙

원칙 1: 불필요한 기술적 필드를 제거하고 의미 있는 정보만 전달합니다. 원칙 2: 수치 데이터에 단위와 맥락을 명시합니다. 원칙 3: 중첩된 JSON은 평탄화(flatten)하여 가독성을 높입니다.

구조화 데이터 → 자연어 변환 패턴

from typing import Any

def flatten_nested_json(data: dict, prefix: str = "", sep: str = ".") -> dict:
    """중첩된 JSON을 평탄화합니다."""
    result = {}
    for key, value in data.items():
        new_key = f"{prefix}{sep}{key}" if prefix else key
        if isinstance(value, dict):
            result.update(flatten_nested_json(value, new_key, sep))
        elif isinstance(value, list) and all(isinstance(i, dict) for i in value):
            for i, item in enumerate(value):
                result.update(flatten_nested_json(item, f"{new_key}[{i}]", sep))
        else:
            result[new_key] = value
    return result

def filter_relevant_fields(
    data: dict,
    relevant_keys: list[str]
) -> dict:
    """필요한 필드만 추출합니다."""
    return {k: v for k, v in data.items() if k in relevant_keys}

def convert_to_natural_language(
    data: dict,
    field_descriptions: dict[str, str]
) -> str:
    """딕셔너리 데이터를 자연어 문장으로 변환합니다."""
    lines = []
    for field, description in field_descriptions.items():
        if field in data and data[field] is not None:
            lines.append(f"- {description}: {data[field]}")
    return "\n".join(lines)

    "weather": [{"description": "맑음"}],
    "wind": {"speed": 3.5, "deg": 180}

# 평탄화

# 자연어 변환

토큰 효율을 위한 데이터 압축 전략

LLM API 호출 비용은 입력 토큰 수에 비례합니다. 대용량 API 응답을 그대로 LLM에 전달하면 비용이 급증할 수 있습니다.

def compress_api_response(
    data: Any,
    max_items: int = 5,
    max_text_length: int = 500
) -> Any:
    """API 응답을 토큰 효율적으로 압축합니다."""
    if isinstance(data, dict):
        # 불필요한 메타데이터 키 제거
        remove_keys = {"id", "url", "urlToImage", "status", "totalResults"}
        return {
            k: compress_api_response(v, max_items, max_text_length)
            for k, v in data.items()
            if k not in remove_keys
        }
    elif isinstance(data, list):
        # 리스트는 max_items 개수로 제한
        return [
            compress_api_response(item, max_items, max_text_length)
            for item in data[:max_items]
        ]
    elif isinstance(data, str) and len(data) > max_text_length:
        # 긴 텍스트는 잘라내기
        return data[:max_text_length] + "..."
    return data

def build_multi_source_context(sources: dict[str, str]) -> str:
    """여러 API 데이터를 하나의 컨텍스트 문자열로 통합합니다."""
    sections = []
    for source_name, content in sources.items():
        sections.append(f"[{source_name}]\n{content}")
    return "\n\n".join(sections)

        news_task = async_get_news(session, keyword)
        youtube_task = async_get_youtube_videos(session, keyword)

        weather, news, videos = await asyncio.gather(
            weather_task, news_task, youtube_task,
            return_exceptions=False
        )
    return weather, news, videos

오류 처리 및 폴백 전략

외부 API는 언제든지 장애가 발생할 수 있습니다. 견고한 시스템을 위해 폴백(fallback) 전략이 필요합니다.

from functools import wraps
import time

def retry_with_backoff(max_retries: int = 3, base_delay: float = 1.0):
    """지수 백오프 재시도 데코레이터"""
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            for attempt in range(max_retries):
                try:
                    return func(*args, **kwargs)
                except requests.exceptions.HTTPError as e:
                    if e.response.status_code == 429:  # Rate Limit
                        delay = base_delay * (2 ** attempt)
                        print(f"Rate limit 도달. {delay}초 후 재시도...")
                        time.sleep(delay)
                    elif attempt == max_retries - 1:
                        raise
                except requests.exceptions.RequestException as e:
                    if attempt == max_retries - 1:
                        raise
                    time.sleep(base_delay * (2 ** attempt))
            return None
        return wrapper
    return decorator

class APIOrchestrator:
    """여러 API를 조율하는 오케스트레이터 클래스"""

    def __init__(self):
        self.cache = {}  # 간단한 인메모리 캐시
        self.cache_ttl = 300  # 5분 캐시

    def _get_cached(self, key: str) -> Optional[Any]:
        if key in self.cache:
            data, timestamp = self.cache[key]
            if time.time() - timestamp < self.cache_ttl:
                return data
        return None

    def _set_cache(self, key: str, data: Any):
        self.cache[key] = (data, time.time())

    @retry_with_backoff(max_retries=3)
    def get_weather_safe(self, city: str) -> Optional[dict]:
        cached = self._get_cached(f"weather_{city}")
        if cached:
            return cached
        data = get_current_weather(city)
        self._set_cache(f"weather_{city}", data)
        return data

    @retry_with_backoff(max_retries=3)
    def get_news_safe(self, keyword: str) -> list[dict]:
        cached = self._get_cached(f"news_{keyword}")
        if cached:
            return cached
        data = search_news(keyword)
        self._set_cache(f"news_{keyword}", data)
        return data

    def get_combined_context(self, city: str, keyword: str) -> str:
        """여러 API 데이터를 통합하여 LLM 컨텍스트를 생성합니다."""
        parts = []

        # 날씨 정보 (실패해도 계속 진행)
        try:
            weather = self.get_weather_safe(city)
            if weather:
                parts.append(f"[날씨]\n{parse_weather_for_llm(weather)}")
        except Exception as e:
            parts.append(f"[날씨]\n날씨 정보를 가져올 수 없습니다. ({e})")

        # 뉴스 정보
        try:
            articles = self.get_news_safe(keyword)
            if articles:
                news_summary = "\n".join([
                    f"- {a['title']} ({a['source']['name']})"
                    for a in articles[:3]
                ])
                parts.append(f"[뉴스]\n{news_summary}")
        except Exception as e:
            parts.append(f"[뉴스]\n뉴스 정보를 가져올 수 없습니다. ({e})")

        return "\n\n".join(parts)

def get_weather_text(city: str) -> str:
    """날씨 데이터를 텍스트로 반환합니다."""
    try:
        url = "https://api.openweathermap.org/data/2.5/weather"

            "units": "metric", "lang": "kr"

            f"기온: {d['main']['temp']}°C (체감 {d['main']['feels_like']}°C), "

def get_news_text(keyword: str, max_articles: int = 5) -> str:
    """뉴스 데이터를 텍스트로 반환합니다."""
    try:
        url = "https://newsapi.org/v2/everything"
        params = {
            "q": keyword, "language": "ko",
            "pageSize": max_articles, "sortBy": "relevancy",
            "apiKey": NEWS_API_KEY
        }
        resp = requests.get(url, params=params, timeout=5)
        resp.raise_for_status()
        articles = resp.json().get("articles", [])
        if not articles:
            return f"'{keyword}' 관련 최근 뉴스 없음"

        lines = []
        for i, a in enumerate(articles[:max_articles], 1):
            title = a.get("title", "제목 없음")
            source = a.get("source", {}).get("name", "")
            published = a.get("publishedAt", "")[:10]
            lines.append(f"{i}. [{source}] {title} ({published})")
        return "\n".join(lines)
    except Exception as e:
        return f"뉴스 조회 실패: {e}"


def get_youtube_text(query: str, max_results: int = 5) -> str:
    """YouTube 검색 결과를 텍스트로 반환합니다."""
    try:
        url = "https://www.googleapis.com/youtube/v3/search"
        params = {
            "part": "snippet", "q": query, "type": "video",
            "maxResults": max_results, "key": YOUTUBE_API_KEY
        }
        resp = requests.get(url, params=params, timeout=5)
        resp.raise_for_status()
        items = resp.json().get("items", [])
        if not items:
            return f"'{query}' 관련 YouTube 영상 없음"

        lines = []
        for i, item in enumerate(items, 1):
            snippet = item["snippet"]
            vid_id = item["id"]["videoId"]
            title = snippet["title"]
            channel = snippet["channelTitle"]
            lines.append(
                f"{i}. {title} - {channel} "
                f"(https://youtu.be/{vid_id})"
            )
        return "\n".join(lines)
    except Exception as e:
        return f"YouTube 조회 실패: {e}"

class IntegratedInfoBot:
    """날씨, 뉴스, YouTube 데이터를 통합하여 답변하는 봇"""

    def __init__(self):

1. 제공된 실제 데이터를 기반으로 답변합니다.

3. 필요시 추가 질문을 통해 정보를 명확히 합니다.

                        "- needs_youtube: YouTube 정보 필요 여부 (true/false)\n"
                        "JSON만 출력하세요."
                    )
                },
                {"role": "user", "content": user_input}
            ],
            response_format={"type": "json_object"}
        )
        import json
        try:
            return json.loads(response.choices[0].message.content)
        except Exception:
            return {
                "city": "Seoul", "keyword": user_input,
                "needs_weather": True, "needs_news": True, "needs_youtube": True

        """엔티티 정보를 바탕으로 API 데이터를 수집합니다."""

        if entities.get("needs_weather", True):
            city = entities.get("city", "Seoul")
            weather_text = get_weather_text(city)
            data_parts.append(f"[실시간 날씨 데이터]\n{weather_text}")

        if entities.get("needs_news", True):
            keyword = entities.get("keyword", "")
            news_text = get_news_text(keyword)
            data_parts.append(f"[최신 뉴스 데이터]\n{news_text}")

        if entities.get("needs_youtube", True):
            keyword = entities.get("keyword", "")
            youtube_text = get_youtube_text(keyword)
            data_parts.append(f"[YouTube 관련 영상]\n{youtube_text}")

        return "\n\n".join(data_parts)

        """사용자 메시지를 받아 통합 데이터 기반 답변을 반환합니다."""
        # 1. 엔티티 추출
        entities = self._extract_entities(user_input)

        # 2. API 데이터 수집
        real_time_data = self._gather_data(entities)

        # 3. 대화 이력에 추가 (데이터 포함)
        augmented_message = (
            f"{user_input}\n\n"
            f"--- 참고할 실시간 데이터 ---\n{real_time_data}"
        )
        self.conversation_history.append({
            "role": "user",
            "content": augmented_message
        })

        # 4. LLM으로 최종 답변 생성
        messages = [
            {"role": "system", "content": self.system_prompt}
        ] + self.conversation_history

        response = client.chat.completions.create(
            model="gpt-4o-mini",
            messages=messages,
            max_tokens=1000
        )

        # 대화 이력에 어시스턴트 응답 저장 (데이터 없는 원본 사용자 메시지로 교체)
        self.conversation_history[-1] = {"role": "user", "content": user_input}
        self.conversation_history.append({
            "role": "assistant",
            "content": assistant_reply
        })

        return assistant_reply

        """대화 이력을 초기화합니다."""
        self.conversation_history = []
        print("대화 이력이 초기화되었습니다.")

18.1 이메일 발송 연동 (SMTP, smtplib)

SMTP와 smtplib 개요

SMTP(Simple Mail Transfer Protocol)는 이메일을 전송하기 위한 표준 프로토콜입니다. 파이썬 표준 라이브러리인 smtplib를 사용하면 별도의 외부 패키지 없이 이메일을 발송할 수 있습니다.

항목 설명
SMTP 이메일 전송 프로토콜 (포트 25, 587, 465)
smtplib 파이썬 표준 SMTP 클라이언트 라이브러리
SSL/TLS 암호화 연결 방식 (포트 465: SSL, 포트 587: STARTTLS)
MIME 첨부 파일, HTML 본문 등을 지원하는 메시지 형식

기본 이메일 발송

import smtplib
from email.mime.text import MIMEText
from email.mime.multipart import MIMEMultipart

def send_email(
    smtp_host: str,
    smtp_port: int,
    username: str,
    password: str,
    to_address: str,
    subject: str,
    body: str,
    html_body: str = None
):
    """기본 이메일 발송 함수"""
    # 멀티파트 메시지 생성
    msg = MIMEMultipart("alternative")
    msg["Subject"] = subject
    msg["From"] = username
    msg["To"] = to_address

    # 텍스트 파트 추가
    text_part = MIMEText(body, "plain", "utf-8")
    msg.attach(text_part)

    # HTML 파트 추가 (선택사항)
    if html_body:
        html_part = MIMEText(html_body, "html", "utf-8")
        msg.attach(html_part)

    # SMTP 서버 연결 및 발송 (STARTTLS 방식)
    with smtplib.SMTP(smtp_host, smtp_port) as server:
        server.ehlo()
        server.starttls()          # TLS 암호화 시작
        server.login(username, password)
        server.sendmail(username, to_address, msg.as_string())

    print(f"이메일 발송 완료: {to_address}")

    msg = MIMEMultipart()
    msg["Subject"] = subject
    msg["From"] = username
    msg["To"] = to_address

    # 본문 추가
    msg.attach(MIMEText(body, "plain", "utf-8"))

    # 첨부 파일 처리
    with open(attachment_path, "rb") as f:
        part = MIMEBase("application", "octet-stream")
        part.set_payload(f.read())

    encoders.encode_base64(part)
    filename = os.path.basename(attachment_path)
    part.add_header(
        "Content-Disposition",
        f"attachment; filename={filename}"
    )
    msg.attach(part)

    with smtplib.SMTP(smtp_host, smtp_port) as server:
        server.starttls()
        server.login(username, password)
        server.sendmail(username, to_address, msg.as_string())

    print(f"첨부 파일 이메일 발송 완료: {filename}")

이메일 발송 유틸리티 클래스

반복적인 이메일 발송을 편리하게 처리하기 위해 클래스로 래핑합니다.

import smtplib
from email.mime.text import MIMEText
from email.mime.multipart import MIMEMultipart
from dataclasses import dataclass
from typing import Optional

@dataclass
class EmailConfig:
    """이메일 설정 데이터 클래스"""
    smtp_host: str
    smtp_port: int
    username: str
    password: str


class EmailSender:
    """이메일 발송 유틸리티 클래스"""

    def __init__(self, config: EmailConfig):
        self.config = config

    def _create_connection(self):
        """SMTP 연결 생성"""
        server = smtplib.SMTP(self.config.smtp_host, self.config.smtp_port)
        server.ehlo()
        server.starttls()
        server.login(self.config.username, self.config.password)
        return server

    def send(
        self,
        to: str | list[str],
        subject: str,
        body: str,
        html: Optional[str] = None,
        cc: Optional[list[str]] = None
    ) -> bool:
        """이메일 발송 메서드"""
        # 수신자 목록 처리
        recipients = [to] if isinstance(to, str) else to

        msg = MIMEMultipart("alternative")
        msg["Subject"] = subject
        msg["From"] = self.config.username
        msg["To"] = ", ".join(recipients)
        if cc:
            msg["Cc"] = ", ".join(cc)
            recipients.extend(cc)

        msg.attach(MIMEText(body, "plain", "utf-8"))
        if html:
            msg.attach(MIMEText(html, "html", "utf-8"))

        try:
            with self._create_connection() as server:
                server.sendmail(
                    self.config.username,
                    recipients,
                    msg.as_string()
                )
            return True
        except smtplib.SMTPException as e:
            print(f"이메일 발송 실패: {e}")
            return False

항상 다음 형식으로 응답하세요:
제목: [제목 내용]
본문: [본문 내용]"""

    user_prompt = f"다음 정보로 이메일을 작성해주세요:\n{context}"

        model="claude-opus-4-5",
        max_tokens=1024,
        system=system_prompt,
        messages=[{"role": "user", "content": user_prompt}]
    )

    lines = text.split("\n")

    body_lines = []

            if rest:
                body_lines.append(rest)
        elif is_body:
            body_lines.append(line)

    return {
        "type": email_type.value,
        "subject": subject,
        "body": "\n".join(body_lines).strip()
    }

18.3 수신 이메일 분석 및 자동 분류

IMAP을 이용한 이메일 수신

IMAP(Internet Message Access Protocol)을 사용하면 서버에서 이메일을 읽고 관리할 수 있습니다. 파이썬 표준 라이브러리의 imaplib을 사용합니다.

import imaplib
import email
from email.header import decode_header
from dataclasses import dataclass
from typing import Optional


@dataclass
class EmailMessage:
    """이메일 메시지 데이터 클래스"""
    uid: str
    subject: str
    sender: str
    date: str
    body: str
    is_read: bool


def decode_str(value: str) -> str:
    """이메일 헤더 디코딩"""
    decoded_parts = decode_header(value)
    result = []
    for part, charset in decoded_parts:
        if isinstance(part, bytes):
            charset = charset or "utf-8"
            result.append(part.decode(charset, errors="replace"))
        else:
            result.append(part)
    return "".join(result)


def fetch_emails(
    imap_host: str,
    username: str,
    password: str,
    folder: str = "INBOX",
    limit: int = 10,
    unread_only: bool = False
) -> list[EmailMessage]:
    """IMAP으로 이메일 수신"""

    messages = []

    with imaplib.IMAP4_SSL(imap_host) as mail:
        mail.login(username, password)
        mail.select(folder)

        # 검색 조건
        search_criteria = "UNSEEN" if unread_only else "ALL"
        _, message_ids = mail.search(None, search_criteria)

        uid_list = message_ids[0].split()
        # 최신 limit개만 가져오기
        uid_list = uid_list[-limit:]

        for uid in reversed(uid_list):
            _, msg_data = mail.fetch(uid, "(RFC822 FLAGS)")

            raw_email = msg_data[0][1]
            flags = msg_data[0][0].decode()
            is_read = "\\Seen" in flags

            msg = email.message_from_bytes(raw_email)

            subject = decode_str(msg.get("Subject", ""))
            sender = decode_str(msg.get("From", ""))
            date = msg.get("Date", "")

            # 본문 추출
            body = ""
            if msg.is_multipart():
                for part in msg.walk():
                    if part.get_content_type() == "text/plain":
                        charset = part.get_content_charset() or "utf-8"
                        body = part.get_payload(decode=True).decode(
                            charset, errors="replace"
                        )
                        break
            else:
                charset = msg.get_content_charset() or "utf-8"
                body = msg.get_payload(decode=True).decode(
                    charset, errors="replace"
                )

            messages.append(EmailMessage(
                uid=uid.decode(),
                subject=subject,
                sender=sender,
                date=date,
                body=body[:500],  # 본문 앞 500자만
                is_read=is_read
            ))

    return messages

APScheduler를 이용한 예약 발송

APScheduler(Advanced Python Scheduler)는 파이썬에서 작업 예약을 위한 강력한 라이브러리입니다.

from apscheduler.schedulers.background import BackgroundScheduler
from apscheduler.triggers.cron import CronTrigger
from apscheduler.triggers.date import DateTrigger
from datetime import datetime, timedelta
import pytz
from dataclasses import dataclass, field
from typing import Callable, Optional
import uuid


@dataclass
class ScheduledEmail:
    """예약 이메일 데이터 클래스"""
    job_id: str
    to: str
    subject: str
    body: str
    scheduled_time: datetime
    status: str = "pending"  # pending, sent, failed, cancelled


class EmailScheduler:
    """이메일 예약 발송 시스템"""

    def __init__(self, email_sender: "EmailSender"):
        self.sender = email_sender
        self.scheduler = BackgroundScheduler(
            timezone=pytz.timezone("Asia/Seoul")
        )
        self.scheduled_jobs: dict[str, ScheduledEmail] = {}
        self.scheduler.start()
        print("이메일 예약 발송 시스템 시작")

    def schedule_once(
        self,
        to: str,
        subject: str,
        body: str,
        send_at: datetime
    ) -> str:
        """특정 시각에 한 번 발송 예약"""
        job_id = str(uuid.uuid4())[:8]

        email_info = ScheduledEmail(
            job_id=job_id,
            to=to,
            subject=subject,
            body=body,
            scheduled_time=send_at
        )
        self.scheduled_jobs[job_id] = email_info

        self.scheduler.add_job(
            func=self._send_job,
            trigger=DateTrigger(run_date=send_at),
            args=[job_id],
            id=job_id
        )

        print(f"이메일 예약 완료 [ID: {job_id}]: {send_at.strftime('%Y-%m-%d %H:%M')}")
        return job_id

    def schedule_recurring(
        self,
        to: str,
        subject_template: str,
        body_generator: Callable[[], str],
        cron_expression: str
    ) -> str:
        """반복 발송 예약 (cron 표현식 사용)"""
        job_id = str(uuid.uuid4())[:8]

        self.scheduler.add_job(
            func=self._send_recurring,
            trigger=CronTrigger.from_crontab(cron_expression),
            args=[to, subject_template, body_generator],
            id=job_id
        )

        print(f"반복 발송 예약 [ID: {job_id}]: {cron_expression}")
        return job_id

    def _send_job(self, job_id: str):
        """예약된 이메일 발송 실행"""
        if job_id not in self.scheduled_jobs:
            return

        job = self.scheduled_jobs[job_id]
        success = self.sender.send(
            to=job.to,
            subject=job.subject,
            body=job.body
        )
        job.status = "sent" if success else "failed"
        print(f"예약 이메일 발송 {'완료' if success else '실패'} [ID: {job_id}]")

    def _send_recurring(
        self,
        to: str,
        subject_template: str,
        body_generator: Callable[[], str]
    ):
        """반복 발송 실행"""
        now = datetime.now(pytz.timezone("Asia/Seoul"))
        subject = subject_template.format(
            date=now.strftime("%Y-%m-%d"),
            time=now.strftime("%H:%M")
        )
        body = body_generator()
        self.sender.send(to=to, subject=subject, body=body)

    def cancel(self, job_id: str) -> bool:
        """예약 취소"""
        try:
            self.scheduler.remove_job(job_id)
            if job_id in self.scheduled_jobs:
                self.scheduled_jobs[job_id].status = "cancelled"
            print(f"예약 취소 완료 [ID: {job_id}]")
            return True
        except Exception as e:
            print(f"예약 취소 실패: {e}")
            return False

    def list_jobs(self) -> list[dict]:
        """예약된 작업 목록 조회"""
        jobs = []
        for job in self.scheduler.get_jobs():
            info = {
                "id": job.id,
                "next_run": str(job.next_run_time)
            }
            if job.id in self.scheduled_jobs:
                email_info = self.scheduled_jobs[job.id]
                info.update({
                    "to": email_info.to,
                    "subject": email_info.subject,
                    "status": email_info.status
                })
            jobs.append(info)
        return jobs

    def shutdown(self):
        """스케줄러 종료"""
        self.scheduler.shutdown()
        print("이메일 예약 발송 시스템 종료")

큐(Queue) 기반 안정적인 발송 시스템

대량 이메일 발송 시에는 Queue를 활용하여 안정적으로 처리합니다.

import queue
import threading
import time
from dataclasses import dataclass
from typing import Optional


@dataclass
class EmailTask:
    """이메일 작업 데이터 클래스"""
    to: str
    subject: str
    body: str
    priority: int = 5  # 1(높음) ~ 10(낮음)
    retry_count: int = 0
    max_retries: int = 3


class EmailQueue:
    """큐 기반 이메일 발송 시스템"""

    def __init__(self, email_sender: "EmailSender", workers: int = 2):
        self.sender = email_sender
        self.queue = queue.PriorityQueue()
        self.workers = workers
        self.running = False
        self._threads: list[threading.Thread] = []

    def enqueue(self, task: EmailTask):
        """이메일 작업을 큐에 추가"""
        # priority 낮을수록 먼저 처리 (PriorityQueue)
        self.queue.put((task.priority, time.time(), task))
        print(f"큐 추가: [{task.priority}순위] {task.subject}")

    def start(self):
        """워커 스레드 시작"""
        self.running = True
        for i in range(self.workers):
            t = threading.Thread(
                target=self._worker,
                name=f"EmailWorker-{i+1}",
                daemon=True
            )
            t.start()
            self._threads.append(t)
        print(f"{self.workers}개 워커 스레드 시작")

    def _worker(self):
        """워커 스레드: 큐에서 작업을 꺼내 처리"""
        while self.running:
            try:
                priority, _, task = self.queue.get(timeout=1)
                self._process_task(task)
                self.queue.task_done()
            except queue.Empty:
                continue

    def _process_task(self, task: EmailTask):
        """이메일 발송 처리 (재시도 포함)"""
        success = self.sender.send(
            to=task.to,
            subject=task.subject,
            body=task.body
        )

        if not success and task.retry_count < task.max_retries:
            task.retry_count += 1
            print(f"재시도 ({task.retry_count}/{task.max_retries}): {task.subject}")
            time.sleep(2 ** task.retry_count)  # 지수 백오프
            self.enqueue(task)

    def stop(self):
        """발송 시스템 종료 (큐 처리 완료 후)"""
        self.queue.join()  # 모든 작업 완료 대기
        self.running = False
        print("이메일 큐 시스템 종료")

class AlertLevel(Enum):
    INFO = "정보"
    WARNING = "주의"
    CRITICAL = "긴급"


@dataclass
class AlertRule:
    """알림 규칙 데이터 클래스"""
    name: str
    condition: Callable[[], bool]
    level: AlertLevel
    recipients: list[str]
    message_template: str
    cooldown_minutes: int = 60  # 최소 알림 간격


class SmartAlertSystem:
    """LLM 기반 스마트 알림 시스템"""

    def __init__(self, email_sender: "EmailSender"):
        self.sender = email_sender
        self.rules: list[AlertRule] = []
        self.last_sent: dict[str, float] = {}

    def add_rule(self, rule: AlertRule):
        """알림 규칙 추가"""
        self.rules.append(rule)
        print(f"알림 규칙 추가: {rule.name} [{rule.level.value}]")

    def _should_send(self, rule_name: str, cooldown: int) -> bool:
        """쿨다운 체크: 최소 발송 간격 확인"""
        last = self.last_sent.get(rule_name, 0)
        elapsed = time.time() - last
        return elapsed > cooldown * 60

    def generate_alert_message(
        self,
        rule: AlertRule,
        context: dict
    ) -> str:
        """LLM으로 상황에 맞는 알림 메시지 생성"""

        prompt = f"""다음 알림 상황에 맞는 이메일 본문을 작성해주세요.

알림 유형: {rule.level.value}
규칙 이름: {rule.name}
템플릿: {rule.message_template}
상황 데이터: {context}

간결하고 명확하게, 즉시 조치해야 할 내용을 강조하여 작성해주세요."""

        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=256,
            messages=[{"role": "user", "content": prompt}]
        )

        return response.content[0].text

    def check_and_notify(self, context: dict = None):
        """모든 규칙 점검 및 필요 시 알림 발송"""
        context = context or {}
        triggered = []

        for rule in self.rules:
            try:
                if rule.condition() and self._should_send(
                    rule.name, rule.cooldown_minutes
                ):
                    message = self.generate_alert_message(rule, context)
                    subject = f"[{rule.level.value}] {rule.name}"

                    for recipient in rule.recipients:
                        self.sender.send(
                            to=recipient,
                            subject=subject,
                            body=message
                        )

                    self.last_sent[rule.name] = time.time()
                    triggered.append(rule.name)
                    print(f"알림 발송: {rule.name} → {rule.recipients}")

            except Exception as e:
                print(f"알림 규칙 처리 오류 ({rule.name}): {e}")

        return triggered
