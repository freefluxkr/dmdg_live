이번 챕터에서는 파이썬과 LLM(Large Language Model)을 결합하여 이메일 발송부터 수신 분류, 예약 발송, 그리고 복잡한 업무 자동화 시나리오까지 구현하는 방법을 학습합니다. 반복적인 이메일 업무를 AI가 처리하도록 자동화함으로써 생산성을 크게 높일 수 있습니다.

이번 챕터를 마치면 LLM이 이메일을 직접 작성하고, 수신 메일을 분류하며, 예약 발송과 업무 보고서 생성까지 자동으로 처리하는 시스템을 직접 구축할 수 있습니다.

18.1 이메일 발송 연동 (SMTP, smtplib)

SMTP와 smtplib 개요

SMTP(Simple Mail Transfer Protocol)는 이메일을 전송하기 위한 표준 프로토콜입니다. 파이썬 표준 라이브러리인 smtplib를 사용하면 별도의 외부 패키지 없이 이메일을 발송할 수 있습니다.

항목 설명
SMTP 이메일 전송 프로토콜 (포트 25, 587, 465)
smtplib 파이썬 표준 SMTP 클라이언트 라이브러리
SSL/TLS 암호화 연결 방식 (포트 465: SSL, 포트 587: STARTTLS)
MIME 첨부 파일, HTML 본문 등을 지원하는 메시지 형식

기본 이메일 발송

import smtplib
from email.mime.text import MIMEText
from email.mime.multipart import MIMEMultipart

def send_email(
    smtp_host: str,
    smtp_port: int,
    username: str,
    password: str,
    to_address: str,
    subject: str,
    body: str,
    html_body: str = None
):
    """기본 이메일 발송 함수"""
    # 멀티파트 메시지 생성
    msg = MIMEMultipart("alternative")
    msg["Subject"] = subject
    msg["From"] = username
    msg["To"] = to_address

    # 텍스트 파트 추가
    text_part = MIMEText(body, "plain", "utf-8")
    msg.attach(text_part)

    # HTML 파트 추가 (선택사항)
    if html_body:
        html_part = MIMEText(html_body, "html", "utf-8")
        msg.attach(html_part)

    # SMTP 서버 연결 및 발송 (STARTTLS 방식)
    with smtplib.SMTP(smtp_host, smtp_port) as server:
        server.ehlo()
        server.starttls()          # TLS 암호화 시작
        server.login(username, password)
        server.sendmail(username, to_address, msg.as_string())

    print(f"이메일 발송 완료: {to_address}")

import smtplib
import os
from email.mime.text import MIMEText
from email.mime.multipart import MIMEMultipart
from email.mime.base import MIMEBase
from email import encoders

def send_email_with_attachment(
    smtp_host: str,
    smtp_port: int,
    username: str,
    password: str,
    to_address: str,
    subject: str,
    body: str,
    attachment_path: str
):
    """첨부 파일을 포함한 이메일 발송"""
    msg = MIMEMultipart()
    msg["Subject"] = subject
    msg["From"] = username
    msg["To"] = to_address

    # 본문 추가
    msg.attach(MIMEText(body, "plain", "utf-8"))

    # 첨부 파일 처리
    with open(attachment_path, "rb") as f:
        part = MIMEBase("application", "octet-stream")
        part.set_payload(f.read())

    encoders.encode_base64(part)
    filename = os.path.basename(attachment_path)
    part.add_header(
        "Content-Disposition",
        f"attachment; filename={filename}"
    )
    msg.attach(part)

    with smtplib.SMTP(smtp_host, smtp_port) as server:
        server.starttls()
        server.login(username, password)
        server.sendmail(username, to_address, msg.as_string())

    print(f"첨부 파일 이메일 발송 완료: {filename}")

이메일 발송 유틸리티 클래스

반복적인 이메일 발송을 편리하게 처리하기 위해 클래스로 래핑합니다.

import smtplib
from email.mime.text import MIMEText
from email.mime.multipart import MIMEMultipart
from dataclasses import dataclass
from typing import Optional

@dataclass
class EmailConfig:
    """이메일 설정 데이터 클래스"""
    smtp_host: str
    smtp_port: int
    username: str
    password: str


class EmailSender:
    """이메일 발송 유틸리티 클래스"""

    def __init__(self, config: EmailConfig):
        self.config = config

    def _create_connection(self):
        """SMTP 연결 생성"""
        server = smtplib.SMTP(self.config.smtp_host, self.config.smtp_port)
        server.ehlo()
        server.starttls()
        server.login(self.config.username, self.config.password)
        return server

    def send(
        self,
        to: str | list[str],
        subject: str,
        body: str,
        html: Optional[str] = None,
        cc: Optional[list[str]] = None
    ) -> bool:
        """이메일 발송 메서드"""
        # 수신자 목록 처리
        recipients = [to] if isinstance(to, str) else to

        msg = MIMEMultipart("alternative")
        msg["Subject"] = subject
        msg["From"] = self.config.username
        msg["To"] = ", ".join(recipients)
        if cc:
            msg["Cc"] = ", ".join(cc)
            recipients.extend(cc)

        msg.attach(MIMEText(body, "plain", "utf-8"))
        if html:
            msg.attach(MIMEText(html, "html", "utf-8"))

        try:
            with self._create_connection() as server:
                server.sendmail(
                    self.config.username,
                    recipients,
                    msg.as_string()
                )
            return True
        except smtplib.SMTPException as e:
            print(f"이메일 발송 실패: {e}")
            return False


# 사용 예시

    smtp_host="smtp.gmail.com",

    username="your_email@gmail.com",

)

18.2 LLM 기반 이메일 내용 자동 생성

LLM을 활용한 이메일 작성의 장점

LLM(Large Language Model)을 이메일 작성에 연동하면 상황에 맞는 자연스러운 이메일을 자동으로 생성할 수 있습니다. 단순 템플릿 방식과 비교하면 다음과 같은 차이가 있습니다.

방식 장점 단점
템플릿 방식 일관성 높음, 비용 없음 유연성 부족, 다양한 상황 대응 어려움
LLM 방식 맥락 이해, 자연스러운 문체, 다국어 지원 API 비용 발생, 응답 시간 소요
하이브리드 방식 두 방식의 장점 결합 구현 복잡도 증가

기본 이메일 생성 시스템

from anthropic import Anthropic

client = Anthropic()

def generate_email(
    purpose: str,
    recipient: str,
    key_points: list[str],
    tone: str = "formal",
    language: str = "Korean"
) -> dict:
    """LLM으로 이메일 내용 생성"""

    points_text = "\n".join(f"- {point}" for point in key_points)

    prompt = f"""다음 조건에 맞는 이메일을 작성해주세요.

발송 목적: {purpose}
수신자: {recipient}
어조: {tone} (formal=공식적, friendly=친근한, urgent=긴급한)
언어: {language}
핵심 내용:
{points_text}

다음 형식으로 반환해주세요:
제목: [이메일 제목]
---
[이메일 본문]"""

    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[{"role": "user", "content": prompt}]
    )

    # 제목과 본문 파싱
    full_text = response.content[0].text
    parts = full_text.split("---", 1)

    subject_line = parts[0].strip()
    subject = subject_line.replace("제목:", "").strip()
    body = parts[1].strip() if len(parts) > 1 else full_text

    return {"subject": subject, "body": body}

from anthropic import Anthropic
from enum import Enum

client = Anthropic()

class EmailType(Enum):
    MEETING_REQUEST = "회의 요청"
    FOLLOW_UP = "후속 조치"
    APOLOGY = "사과"
    INTRODUCTION = "자기소개"
    PROPOSAL = "제안서"
    REPORT = "보고서"


def generate_email_by_type(
    email_type: EmailType,
    context: dict
) -> dict:
    """이메일 유형별 특화된 생성 함수"""

    type_instructions = {
        EmailType.MEETING_REQUEST: "회의 목적, 안건, 희망 일정을 명확히 포함하세요.",
        EmailType.FOLLOW_UP: "이전 논의 내용을 간략히 언급하고 후속 조치 사항을 명시하세요.",
        EmailType.APOLOGY: "진심 어린 사과와 함께 재발 방지 방안을 포함하세요.",
        EmailType.INTRODUCTION: "간결하고 임팩트 있게 본인을 소개하세요.",
        EmailType.PROPOSAL: "문제점, 해결책, 기대효과를 구조적으로 제시하세요.",
        EmailType.REPORT: "핵심 수치와 현황을 요약하고 다음 단계를 명시하세요.",
    }

    system_prompt = f"""당신은 전문적인 비즈니스 이메일 작성 전문가입니다.
이메일 유형: {email_type.value}
작성 지침: {type_instructions[email_type]}

항상 다음 형식으로 응답하세요:
제목: [제목 내용]
본문: [본문 내용]"""

    user_prompt = f"다음 정보로 이메일을 작성해주세요:\n{context}"

    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        system=system_prompt,
        messages=[{"role": "user", "content": user_prompt}]
    )

    text = response.content[0].text
    lines = text.split("\n")

    subject = ""
    body_lines = []
    is_body = False

    for line in lines:
        if line.startswith("제목:"):
            subject = line.replace("제목:", "").strip()
        elif line.startswith("본문:"):
            is_body = True
            rest = line.replace("본문:", "").strip()
            if rest:
                body_lines.append(rest)
        elif is_body:
            body_lines.append(line)

    return {
        "type": email_type.value,
        "subject": subject,
        "body": "\n".join(body_lines).strip()
    }

18.3 수신 이메일 분석 및 자동 분류

IMAP을 이용한 이메일 수신

IMAP(Internet Message Access Protocol)을 사용하면 서버에서 이메일을 읽고 관리할 수 있습니다. 파이썬 표준 라이브러리의 imaplib을 사용합니다.

import imaplib
import email
from email.header import decode_header
from dataclasses import dataclass
from typing import Optional


@dataclass
class EmailMessage:
    """이메일 메시지 데이터 클래스"""
    uid: str
    subject: str
    sender: str
    date: str
    body: str
    is_read: bool


def decode_str(value: str) -> str:
    """이메일 헤더 디코딩"""
    decoded_parts = decode_header(value)
    result = []
    for part, charset in decoded_parts:
        if isinstance(part, bytes):
            charset = charset or "utf-8"
            result.append(part.decode(charset, errors="replace"))
        else:
            result.append(part)
    return "".join(result)


def fetch_emails(
    imap_host: str,
    username: str,
    password: str,
    folder: str = "INBOX",
    limit: int = 10,
    unread_only: bool = False
) -> list[EmailMessage]:
    """IMAP으로 이메일 수신"""

    messages = []

    with imaplib.IMAP4_SSL(imap_host) as mail:
        mail.login(username, password)
        mail.select(folder)

        # 검색 조건
        search_criteria = "UNSEEN" if unread_only else "ALL"
        _, message_ids = mail.search(None, search_criteria)

        uid_list = message_ids[0].split()
        # 최신 limit개만 가져오기
        uid_list = uid_list[-limit:]

        for uid in reversed(uid_list):
            _, msg_data = mail.fetch(uid, "(RFC822 FLAGS)")

            raw_email = msg_data[0][1]
            flags = msg_data[0][0].decode()
            is_read = "\\Seen" in flags

            msg = email.message_from_bytes(raw_email)

            subject = decode_str(msg.get("Subject", ""))
            sender = decode_str(msg.get("From", ""))
            date = msg.get("Date", "")

            # 본문 추출
            body = ""
            if msg.is_multipart():
                for part in msg.walk():
                    if part.get_content_type() == "text/plain":
                        charset = part.get_content_charset() or "utf-8"
                        body = part.get_payload(decode=True).decode(
                            charset, errors="replace"
                        )
                        break
            else:
                charset = msg.get_content_charset() or "utf-8"
                body = msg.get_payload(decode=True).decode(
                    charset, errors="replace"
                )

            messages.append(EmailMessage(
                uid=uid.decode(),
                subject=subject,
                sender=sender,
                date=date,
                body=body[:500],  # 본문 앞 500자만
                is_read=is_read
            ))

    return messages

LLM 기반 이메일 자동 분류

from anthropic import Anthropic
import json
from enum import Enum

client = Anthropic()


class EmailCategory(Enum):
    URGENT = "긴급"
    MEETING = "회의/일정"
    REPORT = "보고/결과"
    REQUEST = "요청/협조"
    INFO = "정보/공지"
    PERSONAL = "개인"
    SPAM = "스팸/광고"
    OTHER = "기타"


def classify_email(email_msg: EmailMessage) -> dict:
    """LLM으로 이메일 분류 및 분석"""

    prompt = f"""다음 이메일을 분석하고 JSON 형식으로 분류 결과를 반환해주세요.

제목: {email_msg.subject}
발신자: {email_msg.sender}
본문 (일부): {email_msg.body[:300]}

다음 JSON 형식으로만 응답해주세요:
{{
    "category": "긴급|회의/일정|보고/결과|요청/협조|정보/공지|개인|스팸/광고|기타",
    "priority": "high|medium|low",
    "summary": "한 문장 요약",
    "action_required": true/false,
    "action_description": "필요한 조치 사항 (없으면 null)",
    "keywords": ["키워드1", "키워드2"]
}}"""

    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=512,
        messages=[{"role": "user", "content": prompt}]
    )

    try:
        result = json.loads(response.content[0].text)
        result["uid"] = email_msg.uid
        result["subject"] = email_msg.subject
        result["sender"] = email_msg.sender
        return result
    except json.JSONDecodeError:
        return {
            "uid": email_msg.uid,
            "subject": email_msg.subject,
            "category": "기타",
            "priority": "low",
            "summary": "분류 실패",
            "action_required": False
        }


def classify_multiple_emails(
    emails: list[EmailMessage]
) -> list[dict]:
    """여러 이메일 일괄 분류"""
    results = []
    for email_msg in emails:
        result = classify_email(email_msg)
        results.append(result)
        print(f"분류 완료: [{result['category']}] {result['subject']}")
    return results


def generate_inbox_summary(classified_emails: list[dict]) -> str:
    """받은 편지함 요약 생성"""

    urgent_count = sum(
        1 for e in classified_emails if e.get("priority") == "high"
    )
    action_needed = [
        e for e in classified_emails if e.get("action_required")
    ]

    summary_data = {
        "총 이메일 수": len(classified_emails),
        "긴급 이메일": urgent_count,
        "조치 필요": len(action_needed),
        "분류별 현황": {}
    }

    for email_info in classified_emails:
        cat = email_info.get("category", "기타")

조치가 필요한 이메일:
{chr(10).join([f'- [{e["priority"]}] {e["subject"]}: {e.get("action_description", "")}' for e in action_needed])}

3~5문장으로 간결하게 작성해주세요."""

    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=512,
        messages=[{"role": "user", "content": prompt}]

    return response.content[0].text

APScheduler를 이용한 예약 발송

APScheduler(Advanced Python Scheduler)는 파이썬에서 작업 예약을 위한 강력한 라이브러리입니다.

from apscheduler.schedulers.background import BackgroundScheduler
from apscheduler.triggers.cron import CronTrigger
from apscheduler.triggers.date import DateTrigger
from datetime import datetime, timedelta
import pytz
from dataclasses import dataclass, field
from typing import Callable, Optional
import uuid


@dataclass
class ScheduledEmail:
    """예약 이메일 데이터 클래스"""
    job_id: str
    to: str
    subject: str
    body: str
    scheduled_time: datetime
    status: str = "pending"  # pending, sent, failed, cancelled


class EmailScheduler:
    """이메일 예약 발송 시스템"""

    def __init__(self, email_sender: "EmailSender"):
        self.sender = email_sender
        self.scheduler = BackgroundScheduler(
            timezone=pytz.timezone("Asia/Seoul")
        )
        self.scheduled_jobs: dict[str, ScheduledEmail] = {}
        self.scheduler.start()
        print("이메일 예약 발송 시스템 시작")

    def schedule_once(
        self,
        to: str,
        subject: str,
        body: str,
        send_at: datetime
    ) -> str:
        """특정 시각에 한 번 발송 예약"""
        job_id = str(uuid.uuid4())[:8]

        email_info = ScheduledEmail(
            job_id=job_id,
            to=to,
            subject=subject,
            body=body,
            scheduled_time=send_at
        )
        self.scheduled_jobs[job_id] = email_info

        self.scheduler.add_job(
            func=self._send_job,
            trigger=DateTrigger(run_date=send_at),
            args=[job_id],
            id=job_id
        )

        print(f"이메일 예약 완료 [ID: {job_id}]: {send_at.strftime('%Y-%m-%d %H:%M')}")
        return job_id

    def schedule_recurring(
        self,
        to: str,
        subject_template: str,
        body_generator: Callable[[], str],
        cron_expression: str
    ) -> str:
        """반복 발송 예약 (cron 표현식 사용)"""
        job_id = str(uuid.uuid4())[:8]

        self.scheduler.add_job(
            func=self._send_recurring,
            trigger=CronTrigger.from_crontab(cron_expression),
            args=[to, subject_template, body_generator],
            id=job_id
        )

        print(f"반복 발송 예약 [ID: {job_id}]: {cron_expression}")
        return job_id

    def _send_job(self, job_id: str):
        """예약된 이메일 발송 실행"""
        if job_id not in self.scheduled_jobs:
            return

        job = self.scheduled_jobs[job_id]
        success = self.sender.send(
            to=job.to,
            subject=job.subject,
            body=job.body
        )
        job.status = "sent" if success else "failed"
        print(f"예약 이메일 발송 {'완료' if success else '실패'} [ID: {job_id}]")

    def _send_recurring(
        self,
        to: str,
        subject_template: str,
        body_generator: Callable[[], str]
    ):
        """반복 발송 실행"""
        now = datetime.now(pytz.timezone("Asia/Seoul"))
        subject = subject_template.format(
            date=now.strftime("%Y-%m-%d"),
            time=now.strftime("%H:%M")
        )
        body = body_generator()
        self.sender.send(to=to, subject=subject, body=body)

    def cancel(self, job_id: str) -> bool:
        """예약 취소"""
        try:
            self.scheduler.remove_job(job_id)
            if job_id in self.scheduled_jobs:
                self.scheduled_jobs[job_id].status = "cancelled"
            print(f"예약 취소 완료 [ID: {job_id}]")
            return True
        except Exception as e:
            print(f"예약 취소 실패: {e}")
            return False

    def list_jobs(self) -> list[dict]:
        """예약된 작업 목록 조회"""
        jobs = []
        for job in self.scheduler.get_jobs():
            info = {
                "id": job.id,
                "next_run": str(job.next_run_time)
            }
            if job.id in self.scheduled_jobs:
                email_info = self.scheduled_jobs[job.id]
                info.update({
                    "to": email_info.to,
                    "subject": email_info.subject,
                    "status": email_info.status
                })
            jobs.append(info)
        return jobs

    def shutdown(self):
        """스케줄러 종료"""
        self.scheduler.shutdown()
        print("이메일 예약 발송 시스템 종료")

큐(Queue) 기반 안정적인 발송 시스템

대량 이메일 발송 시에는 Queue를 활용하여 안정적으로 처리합니다.

import queue
import threading
import time
from dataclasses import dataclass
from typing import Optional


@dataclass
class EmailTask:
    """이메일 작업 데이터 클래스"""
    to: str
    subject: str
    body: str
    priority: int = 5  # 1(높음) ~ 10(낮음)
    retry_count: int = 0
    max_retries: int = 3


class EmailQueue:
    """큐 기반 이메일 발송 시스템"""

    def __init__(self, email_sender: "EmailSender", workers: int = 2):
        self.sender = email_sender
        self.queue = queue.PriorityQueue()
        self.workers = workers
        self.running = False
        self._threads: list[threading.Thread] = []

    def enqueue(self, task: EmailTask):
        """이메일 작업을 큐에 추가"""
        # priority 낮을수록 먼저 처리 (PriorityQueue)
        self.queue.put((task.priority, time.time(), task))
        print(f"큐 추가: [{task.priority}순위] {task.subject}")

    def start(self):
        """워커 스레드 시작"""
        self.running = True
        for i in range(self.workers):
            t = threading.Thread(
                target=self._worker,
                name=f"EmailWorker-{i+1}",
                daemon=True
            )
            t.start()
            self._threads.append(t)
        print(f"{self.workers}개 워커 스레드 시작")

    def _worker(self):
        """워커 스레드: 큐에서 작업을 꺼내 처리"""
        while self.running:
            try:
                priority, _, task = self.queue.get(timeout=1)
                self._process_task(task)
                self.queue.task_done()
            except queue.Empty:
                continue

    def _process_task(self, task: EmailTask):
        """이메일 발송 처리 (재시도 포함)"""
        success = self.sender.send(
            to=task.to,
            subject=task.subject,
            body=task.body
        )

        if not success and task.retry_count < task.max_retries:
            task.retry_count += 1
            print(f"재시도 ({task.retry_count}/{task.max_retries}): {task.subject}")
            time.sleep(2 ** task.retry_count)  # 지수 백오프
            self.enqueue(task)

    def stop(self):
        """발송 시스템 종료 (큐 처리 완료 후)"""
        self.queue.join()  # 모든 작업 완료 대기
        self.running = False
        print("이메일 큐 시스템 종료")

        max_tokens=512,
        messages=[{"role": "user", "content": prompt}]

알림 시스템 구현

from anthropic import Anthropic
from dataclasses import dataclass
from enum import Enum
from typing import Callable
import threading
import time

client = Anthropic()


class AlertLevel(Enum):
    INFO = "정보"
    WARNING = "주의"
    CRITICAL = "긴급"


@dataclass
class AlertRule:
    """알림 규칙 데이터 클래스"""
    name: str
    condition: Callable[[], bool]
    level: AlertLevel
    recipients: list[str]
    message_template: str
    cooldown_minutes: int = 60  # 최소 알림 간격


class SmartAlertSystem:
    """LLM 기반 스마트 알림 시스템"""

    def __init__(self, email_sender: "EmailSender"):
        self.sender = email_sender
        self.rules: list[AlertRule] = []
        self.last_sent: dict[str, float] = {}

    def add_rule(self, rule: AlertRule):
        """알림 규칙 추가"""
        self.rules.append(rule)
        print(f"알림 규칙 추가: {rule.name} [{rule.level.value}]")

    def _should_send(self, rule_name: str, cooldown: int) -> bool:
        """쿨다운 체크: 최소 발송 간격 확인"""
        last = self.last_sent.get(rule_name, 0)
        elapsed = time.time() - last
        return elapsed > cooldown * 60

    def generate_alert_message(
        self,
        rule: AlertRule,
        context: dict
    ) -> str:
        """LLM으로 상황에 맞는 알림 메시지 생성"""

        prompt = f"""다음 알림 상황에 맞는 이메일 본문을 작성해주세요.

알림 유형: {rule.level.value}
규칙 이름: {rule.name}
템플릿: {rule.message_template}
상황 데이터: {context}

간결하고 명확하게, 즉시 조치해야 할 내용을 강조하여 작성해주세요."""

        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=256,
            messages=[{"role": "user", "content": prompt}]
        )

        return response.content[0].text

    def check_and_notify(self, context: dict = None):
        """모든 규칙 점검 및 필요 시 알림 발송"""
        context = context or {}
        triggered = []

        for rule in self.rules:
            try:
                if rule.condition() and self._should_send(
                    rule.name, rule.cooldown_minutes
                ):
                    message = self.generate_alert_message(rule, context)
                    subject = f"[{rule.level.value}] {rule.name}"

                    for recipient in rule.recipients:
                        self.sender.send(
                            to=recipient,
                            subject=subject,
                            body=message
                        )

                    self.last_sent[rule.name] = time.time()
                    triggered.append(rule.name)
                    print(f"알림 발송: {rule.name} → {rule.recipients}")

            except Exception as e:
                print(f"알림 규칙 처리 오류 ({rule.name}): {e}")

        return triggered

class AIEmailAutomationSystem:
    """AI 기반 이메일 자동화 통합 시스템"""

    def __init__(self, email_config: "EmailConfig"):
        self.config = email_config
        self.sender = EmailSender(email_config)
        self.scheduler = EmailScheduler(self.sender)
        self.running = False
        self.processed_uids: set[str] = set()

        # 자동 응답 규칙
        self.auto_reply_rules: list[dict] = []

    def add_auto_reply_rule(
        self,
        trigger_keywords: list[str],
        reply_template: str,
        category_filter: Optional[str] = None
    ):
        """자동 응답 규칙 추가"""
        self.auto_reply_rules.append({
            "keywords": [kw.lower() for kw in trigger_keywords],
            "template": reply_template,
            "category": category_filter
        })

    def _should_auto_reply(
        self,
        email_info: dict,
        classified: dict
    ) -> Optional[str]:
        """자동 응답 여부 및 내용 결정"""
        subject_lower = email_info.get("subject", "").lower()
        body_lower = email_info.get("body", "").lower()

        for rule in self.auto_reply_rules:
            # 카테고리 필터 확인
            if rule["category"] and classified.get("category") != rule["category"]:
                continue

            # 키워드 확인
            for keyword in rule["keywords"]:
                if keyword in subject_lower or keyword in body_lower:
                    return rule["template"]

        return None

    def _generate_auto_reply(
        self,
        original_email: "EmailMessage",
        template: str
    ) -> str:
        """LLM으로 자동 응답 생성"""
        prompt = f"""다음 이메일에 대한 자동 응답을 작성해주세요.

원본 이메일:
제목: {original_email.subject}
발신자: {original_email.sender}
내용: {original_email.body[:200]}

응답 템플릿: {template}

자연스럽고 전문적인 응답 이메일 본문을 작성해주세요."""

        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=512,
            messages=[{"role": "user", "content": prompt}]
        )

        return response.content[0].text

    def process_inbox(
        self,
        imap_host: str,
        check_interval: int = 300
    ):
        """받은 편지함 주기적 처리"""
        print(f"받은 편지함 모니터링 시작 (간격: {check_interval}초)")

        while self.running:
            try:
                # 새 이메일 수신
                new_emails = fetch_emails(
                    imap_host=imap_host,
                    username=self.config.username,
                    password=self.config.password,
                    unread_only=True,
                    limit=20
                )

                for email_msg in new_emails:
                    if email_msg.uid in self.processed_uids:
                        continue

                    # LLM 분류
                    classified = classify_email(email_msg)
                    print(
                        f"[{classified['category']}][{classified['priority']}] "
                        f"{email_msg.subject}"
                    )

                    # 긴급 이메일 즉시 알림
                    if classified.get("priority") == "high":
                        self.sender.send(
                            to=self.config.username,
                            subject=f"[긴급 알림] {email_msg.subject}",
                            body=f"긴급 이메일 수신:\n\n{classified['summary']}\n\n"
                                 f"발신자: {email_msg.sender}"
                        )

                    # 자동 응답 확인
                    reply_template = self._should_auto_reply(
                        {"subject": email_msg.subject, "body": email_msg.body},
                        classified
                    )

                    if reply_template:
                        reply_body = self._generate_auto_reply(
                            email_msg, reply_template
                        )
                        # 발신자 주소 추출 (간단화)
                        sender_email = email_msg.sender.split("<")[-1].rstrip(">")
                        self.sender.send(
                            to=sender_email,
                            subject=f"Re: {email_msg.subject}",
                            body=reply_body
                        )
                        print(f"자동 응답 발송: {sender_email}")

                    self.processed_uids.add(email_msg.uid)

            except Exception as e:
                print(f"받은 편지함 처리 오류: {e}")

            time.sleep(check_interval)

    def schedule_daily_report(
        self,
        report_recipient: str,
        report_generator: callable,
        send_hour: int = 9
    ):
        """매일 정해진 시각에 보고서 발송 예약"""
        def generate_and_send():
            report = report_generator()
            self.sender.send(
                to=report_recipient,
                subject=report["title"],
                body=report["content"]
            )

        # 매일 지정 시각에 실행 (APScheduler cron)
        self.scheduler.schedule_recurring(
            to=report_recipient,
            subject_template="일일 보고서 ({date})",
            body_generator=lambda: report_generator()["content"],
            cron_expression=f"0 {send_hour} * * *"
        )
        print(f"일일 보고서 예약: 매일 {send_hour}시")

    def start(self, imap_host: str):
        """시스템 시작"""
        self.running = True
        inbox_thread = threading.Thread(
            target=self.process_inbox,
            args=(imap_host,),
            daemon=True
        )
        inbox_thread.start()
        print("AI 이메일 자동화 시스템 가동 완료")

    def stop(self):
        """시스템 종료"""
        self.running = False
        self.scheduler.shutdown()
        print("AI 이메일 자동화 시스템 종료")
