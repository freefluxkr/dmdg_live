LLM 서비스 레이어

LLM 서비스(LLM Service)는 외부 AI API 호출을 추상화하여 라우트 코드가 특정 제공자에 종속되지 않도록 합니다.

# services/llm_service.py
from openai import OpenAI
from config import Config

client = OpenAI(api_key=Config.OPENAI_API_KEY)

def ask_llm(system_prompt: str, user_message: str, history: list = None) -> str:
    """LLM에 질문하고 응답 텍스트를 반환합니다."""
    messages = [{"role": "system", "content": system_prompt}]

    if history:
        # 최근 10턴만 컨텍스트로 전달 (토큰 절약)
        for turn in history[-10:]:
            messages.append({"role": turn["role"], "content": turn["content"]})

    messages.append({"role": "user", "content": user_message})

    response = client.chat.completions.create(
        model="gpt-4o-mini",
        messages=messages,
        max_tokens=1024,
        temperature=0.7,
    )
    return response.choices[0].message.content

엑셀 서비스 구현

pandas와 openpyxl을 사용하여 업로드된 파일을 파싱하고, 데이터 요약을 LLM에게 전달합니다.

# services/excel_service.py
import pandas as pd
import io
from services.llm_service import ask_llm

EXCEL_SYSTEM_PROMPT = """당신은 데이터 분석 전문가입니다.
주어진 데이터 요약을 바탕으로 핵심 트렌드, 이상값, 비즈니스 인사이트를 제공합니다."""

def analyze_excel(file_bytes: bytes, filename: str, user_question: str) -> dict:
    """엑셀 파일을 분석하고 AI 인사이트를 반환합니다."""
    try:
        df = pd.read_excel(io.BytesIO(file_bytes))
    except Exception:
        df = pd.read_csv(io.BytesIO(file_bytes))

    # 데이터 요약 생성
    summary = _build_summary(df)

    prompt = f"""다음 데이터를 분석해 주세요.

[데이터 개요]
{summary}

[사용자 질문]
{user_question}
"""
    answer = ask_llm(EXCEL_SYSTEM_PROMPT, prompt)
    return {
        "answer": answer,
        "rows": len(df),
        "columns": list(df.columns),
        "summary": summary,
    }

def _build_summary(df: pd.DataFrame) -> str:
    """데이터프레임 요약 문자열을 생성합니다."""
    lines = [
        f"행 수: {len(df)}, 열 수: {len(df.columns)}",
        f"컬럼: {', '.join(df.columns.astype(str))}",
        "",
        "수치형 통계:",
        df.describe().to_string(),
        "",
        "결측값:",
        df.isnull().sum().to_string(),
        "",
        "처음 3행 미리보기:",
        df.head(3).to_string(),
    ]
    return "\n".join(lines)

22.5 검색 기능 통합 (웹검색 + DB검색)

하이브리드 검색 전략

하이브리드 검색(Hybrid Search)은 실시간 웹 검색 결과와 내부 DB 검색 결과를 합쳐 LLM에게 컨텍스트로 제공하는 RAG(Retrieval-Augmented Generation) 패턴입니다.

검색 유형 도구 특징
웹 검색 SerpAPI / Tavily 최신 정보, 실시간
DB 검색 SQLite FTS5 내부 문서, 빠른 속도
결합 LLM 요약 두 결과를 통합 답변

hybrid_search()는 사용자 쿼리를 받아 web_search()(SerpAPI/Tavily)와 db_search()(SQLite FTS5)를 병렬로 호출하고, 두 결과를 묶어 LLM에게 컨텍스트로 전달합니다. 최신성(웹)과 정확성·속도(DB)를 동시에 확보해 환각을 줄이는 것이 핵심입니다.

검색 서비스 구현

# services/search_service.py
import requests
import sqlite3
from services.llm_service import ask_llm
from config import Config

SEARCH_SYSTEM_PROMPT = """당신은 검색 결과를 분석하는 전문가입니다.
웹 검색 결과와 내부 문서를 종합하여 정확하고 신뢰할 수 있는 답변을 제공합니다.
출처를 명시하고, 최신 정보를 우선합니다."""

def web_search(query: str, num_results: int = 5) -> list[dict]:
    """SerpAPI를 사용한 웹 검색입니다."""
    params = {
        "q": query,
        "api_key": Config.SERPAPI_KEY,
        "num": num_results,
        "hl": "ko",
    }
    resp = requests.get("https://serpapi.com/search", params=params, timeout=10)
    resp.raise_for_status()
    data = resp.json()

    results = []
    for item in data.get("organic_results", []):
        results.append({
            "title": item.get("title", ""),
            "snippet": item.get("snippet", ""),
            "link": item.get("link", ""),
            "source": "web",
        })
    return results

def db_search(query: str, db_path: str = "app.db") -> list[dict]:
    """SQLite FTS5 전문 검색입니다."""
    conn = sqlite3.connect(db_path)
    conn.row_factory = sqlite3.Row
    try:
        rows = conn.execute(
            "SELECT title, content, created_at FROM documents_fts WHERE documents_fts MATCH ? LIMIT 5",
            (query,),
        ).fetchall()
        return [
            {"title": r["title"], "snippet": r["content"][:300], "source": "db"}
            for r in rows
        ]
    except sqlite3.OperationalError:
        # FTS 테이블이 없는 경우 빈 결과 반환
        return []
    finally:
        conn.close()

def hybrid_search(query: str) -> dict:
    """웹 검색 + DB 검색 결과를 LLM으로 통합합니다."""
    web_results = web_search(query)
    db_results = db_search(query)
    all_results = web_results + db_results

    if not all_results:
        return {"answer": "검색 결과를 찾을 수 없습니다.", "sources": []}

    context = "\n\n".join(
        f"[{r['source'].upper()}] {r['title']}\n{r['snippet']}"
        for r in all_results
    )
    prompt = f"""다음 검색 결과를 바탕으로 질문에 답변해 주세요.

[검색 결과]
{context}

[질문]
{query}
"""
    answer = ask_llm(SEARCH_SYSTEM_PROMPT, prompt)
    return {
        "answer": answer,
        "sources": [{"title": r["title"], "link": r.get("link", ""), "source": r["source"]} for r in all_results],
    }

let attachedFile = null;

document.getElementById("file-input").addEventListener("change", function () {

document.getElementById("user-input").addEventListener("keydown", function (e) {

  }

async function sendMessage() {
  const input = document.getElementById("user-input");
  const message = input.value.trim();
  if (!message && !attachedFile) return;

  appendMessage("user", message || `[파일 분석] ${attachedFile.name}`);
  input.value = "";
  setLoading(true);

  try {
    let data;
    if (attachedFile) {
      data = await analyzeFile(attachedFile, message || "이 데이터를 분석해 주세요.");
      attachedFile = null;
      document.getElementById("file-label").textContent = "📎 엑셀 파일 첨부";
    } else {
      data = await chatWithAI(message);
    }
    appendMessage("assistant", data.answer || data.error);
  } catch (err) {
    appendMessage("assistant", "오류가 발생했습니다. 다시 시도해 주세요.");
  } finally {
    setLoading(false);
  }
}

async function chatWithAI(message) {
  const resp = await fetch("/api/chat", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ message }),
  });
  return resp.json();
}

async function analyzeFile(file, question) {
  const formData = new FormData();
  formData.append("file", file);
  formData.append("question", question);
  const resp = await fetch("/api/analyze", { method: "POST", body: formData });
  return resp.json();
}

function appendMessage(role, text) {
  const container = document.getElementById("chat-messages");
  const div = document.createElement("div");
  div.className = `message ${role}`;
  div.innerHTML = marked.parse(text); // marked.js로 마크다운 렌더링
  container.appendChild(div);
  container.scrollTop = container.scrollHeight;
}

  document.getElementById("send-btn").disabled = state;
  document.getElementById("send-btn").textContent = state ? "처리 중..." : "전송";
}

보안은 단일 방어선이 아니라 다층 방어(Defense in Depth) 로 설계합니다. 외부 공격이 Rate Limiting → CSRF → 입력 검증 → 비밀(Secret) 격리 4개 레이어를 순차적으로 통과해야 핵심 비즈니스 로직과 LLM API에 닿을 수 있도록 구성하면, 한 레이어가 뚫려도 전체 시스템이 즉시 무너지지 않습니다.

입력 검증 미들웨어

입력 검증(Input Validation)은 XSS(Cross-Site Scripting), 프롬프트 인젝션(Prompt Injection), 과도한 API 사용을 막는 핵심 방어선입니다.

# app.py 추가 — 요청 전 공통 검증
import re
from functools import wraps
from markupsafe import escape

# 허용되지 않는 패턴 (프롬프트 인젝션 시도)
INJECTION_PATTERNS = [
    r"ignore previous instructions",
    r"system prompt",
    r"act as",
    r"jailbreak",
]

def validate_input(text: str, max_length: int = 2000) -> tuple[bool, str]:
    """입력 텍스트를 검증하고 (유효여부, 오류메시지)를 반환합니다."""
    if not text or not text.strip():
        return False, "입력값이 비어 있습니다."
    if len(text) > max_length:
        return False, f"입력값은 {max_length}자 이하여야 합니다."

    lower = text.lower()
    for pattern in INJECTION_PATTERNS:
        if re.search(pattern, lower):
            return False, "허용되지 않는 표현이 포함되어 있습니다."

    return True, ""

def require_json(f):
    """JSON Content-Type을 강제하는 데코레이터입니다."""
    @wraps(f)
    def decorated(*args, **kwargs):
        if not request.is_json:
            return jsonify({"error": "Content-Type은 application/json이어야 합니다."}), 415
        return f(*args, **kwargs)
    return decorated

# app.py (최종 통합본)
from flask import Flask, request, jsonify, render_template, session
from flask_wtf.csrf import CSRFProtect
from flask_limiter import Limiter
from flask_limiter.util import get_remote_address
from werkzeug.utils import secure_filename
from markupsafe import escape
import markdown
import uuid
import time

from config import Config
from services.llm_service import ask_llm
from services.search_service import hybrid_search
from services.excel_service import analyze_excel

app = Flask(__name__)
app.config.from_object(Config)
csrf = CSRFProtect(app)
limiter = Limiter(app=app, key_func=get_remote_address,
                  default_limits=["200 per day", "50 per hour"])

BUSINESS_SYSTEM_PROMPT = """당신은 비즈니스 분석 전문가입니다.
데이터 기반의 명확하고 실행 가능한 인사이트를 제공합니다."""

def allowed_file(filename):
    return "." in filename and \
           filename.rsplit(".", 1)[1].lower() in Config.ALLOWED_EXTENSIONS

@app.route("/")
def index():
    if "session_id" not in session:
        session["session_id"] = str(uuid.uuid4())
        session["history"] = []
    return render_template("index.html")

@app.route("/api/chat", methods=["POST"])
@limiter.limit("20 per minute")
def chat():
    data = request.get_json()
    message = data.get("message", "").strip()
    if not message:
        return jsonify({"error": "메시지를 입력해 주세요."}), 400
    if len(message) > 2000:
        return jsonify({"error": "메시지는 2000자 이하여야 합니다."}), 400

    history = session.get("history", [])
    answer = ask_llm(BUSINESS_SYSTEM_PROMPT, message, history)
    history.append({"role": "user", "content": message})
    history.append({"role": "assistant", "content": answer})
    session["history"] = history
    return jsonify({"answer": answer})

@app.route("/api/analyze", methods=["POST"])
@limiter.limit("10 per minute")
def analyze():
    if "file" not in request.files:
        return jsonify({"error": "파일이 없습니다."}), 400
    file = request.files["file"]
    question = request.form.get("question", "이 데이터의 핵심 인사이트를 알려주세요.")

    if not allowed_file(file.filename):
        return jsonify({"error": "xlsx, xls, csv 파일만 허용됩니다."}), 400

    file_bytes = file.read()
    if len(file_bytes) > Config.MAX_UPLOAD_SIZE_MB * 1024 * 1024:
        return jsonify({"error": f"{Config.MAX_UPLOAD_SIZE_MB}MB 이하 파일만 허용됩니다."}), 413

    start = time.time()
    result = analyze_excel(file_bytes, secure_filename(file.filename), question)
    result["elapsed"] = round(time.time() - start, 2)
    return jsonify(result)

@app.route("/api/search", methods=["POST"])
@limiter.limit("15 per minute")
def search():
    data = request.get_json()
    query = data.get("query", "").strip()
    if not query:
        return jsonify({"error": "검색어를 입력해 주세요."}), 400
    if len(query) > 500:
        return jsonify({"error": "검색어는 500자 이하여야 합니다."}), 400
    result = hybrid_search(query)
    return jsonify(result)

@app.route("/api/clear", methods=["POST"])
def clear_history():
    session["history"] = []
    return jsonify({"status": "ok"})

@app.route("/result")
def result_page():
    answer = request.args.get("answer", "")
    result_html = markdown.markdown(
        answer, extensions=["fenced_code", "tables", "nl2br"]
    )
    return render_template("result.html", result_html=result_html,
                           analysis_type=request.args.get("type", "일반 분석"),
                           elapsed_time=request.args.get("elapsed", "-"),
                           model_name="GPT-4o mini")

if __name__ == "__main__":
    app.run(debug=False)  # 운영 환경에서는 debug=False 필수
