이 챕터에서는 실제 세계의 데이터를 LLM과 연결하는 핵심 기술인 외부 API 연동을 다룹니다. 날씨, 뉴스, YouTube 등 다양한 공개 API를 통해 실시간 데이터를 수집하고, 이를 자연어로 가공하여 사용자에게 의미 있는 정보를 제공하는 방법을 학습합니다.

이전 챕터(Ch.16)에서 Function Calling을 통해 LLM이 외부 함수를 호출하는 구조를 익혔습니다. 이 챕터에서는 그 구조 위에 실제 외부 API를 연동하여 실용적인 AI 애플리케이션을 완성합니다.

17.1 날씨 API 연동 (OpenWeatherMap)

OpenWeatherMap은 전 세계 날씨 데이터를 제공하는 대표적인 무료/유료 기상 API입니다. 무료 플랜만으로도 현재 날씨, 5일 예보, 대기질 정보 등을 조회할 수 있어 학습 및 소규모 프로젝트에 적합합니다.

API 키 발급

1단계: https://openweathermap.org 에 접속하여 회원 가입합니다.

2단계: 로그인 후 상단 메뉴 API keys 탭에서 새 키를 생성합니다. 키 생성 후 활성화까지 최대 2시간이 소요될 수 있습니다.

3단계: 발급받은 키를 환경 변수에 저장합니다.

# .env 파일
OPENWEATHER_API_KEY=your_api_key_here

현재 날씨 데이터 조회

Current Weather API의 엔드포인트는 https://api.openweathermap.org/data/2.5/weather 이며, 도시명 또는 위도/경도로 조회할 수 있습니다.

import os
import requests
from dotenv import load_dotenv

load_dotenv()

OPENWEATHER_API_KEY = os.getenv("OPENWEATHER_API_KEY")
BASE_URL = "https://api.openweathermap.org/data/2.5"

def get_current_weather(city: str, lang: str = "kr") -> dict:
    """도시명으로 현재 날씨를 조회합니다."""
    url = f"{BASE_URL}/weather"
    params = {
        "q": city,
        "appid": OPENWEATHER_API_KEY,
        "units": "metric",   # 섭씨 온도
        "lang": lang         # 한국어 설명
    }
    response = requests.get(url, params=params)
    response.raise_for_status()
    return response.json()

def get_forecast(city: str, days: int = 5) -> dict:
    """5일간 3시간 단위 예보를 조회합니다."""
    url = f"{BASE_URL}/forecast"
    params = {
        "q": city,
        "appid": OPENWEATHER_API_KEY,
        "units": "metric",
        "lang": "kr",
        "cnt": days * 8   # 하루 8개 시간대 (3시간 단위)
    }
    response = requests.get(url, params=params)
    response.raise_for_status()
    return response.json()

17.2 뉴스 API 연동 (NewsAPI)

NewsAPI는 전 세계 7만여 개 뉴스 소스에서 기사를 수집할 수 있는 서비스입니다. 키워드 검색, 카테고리 필터링, 국가별 뉴스 조회 등 다양한 기능을 제공합니다.

API 키 발급

https://newsapi.org 에서 무료 계정을 생성하면 즉시 API 키를 발급받을 수 있습니다. 무료 플랜은 개발자 테스트 용도로 하루 100건 요청이 가능합니다.

# .env 파일에 추가
NEWS_API_KEY=your_newsapi_key_here

키워드 기반 뉴스 수집

import os
import requests
from datetime import datetime, timedelta
from dotenv import load_dotenv

load_dotenv()

NEWS_API_KEY = os.getenv("NEWS_API_KEY")
NEWS_BASE_URL = "https://newsapi.org/v2"

def search_news(
    keyword: str,
    language: str = "ko",
    days_ago: int = 3,
    max_articles: int = 10
) -> list[dict]:
    """키워드로 최근 뉴스를 검색합니다."""
    from_date = (datetime.now() - timedelta(days=days_ago)).strftime("%Y-%m-%d")

    url = f"{NEWS_BASE_URL}/everything"
    params = {
        "q": keyword,
        "language": language,
        "from": from_date,
        "sortBy": "relevancy",  # relevancy, popularity, publishedAt
        "pageSize": max_articles,
        "apiKey": NEWS_API_KEY
    }
    response = requests.get(url, params=params)
    response.raise_for_status()

    data = response.json()
    return data.get("articles", [])

def get_top_headlines(
    country: str = "kr",
    category: str = None,
    max_articles: int = 10
) -> list[dict]:
    """국가 및 카테고리별 헤드라인 뉴스를 가져옵니다."""
    url = f"{NEWS_BASE_URL}/top-headlines"
    params = {
        "country": country,
        "pageSize": max_articles,
        "apiKey": NEWS_API_KEY
    }
    if category:
        # business, entertainment, general, health, science, sports, technology
        params["category"] = category

    response = requests.get(url, params=params)
    response.raise_for_status()

    data = response.json()
    return data.get("articles", [])

# 사용 예시
articles = search_news("인공지능", days_ago=3, max_articles=5)
for article in articles:
    print(f"제목: {article['title']}")
    print(f"출처: {article['source']['name']}")
    print(f"발행일: {article['publishedAt'][:10]}")
    print(f"URL: {article['url']}")
    print("---")

뉴스 요약 자동화

여러 기사를 LLM으로 일괄 요약하는 파이프라인을 구성합니다.

from openai import OpenAI

client = OpenAI()

def summarize_single_article(article: dict) -> str:
    """단일 기사를 요약합니다."""
    title = article.get("title", "제목 없음")
    description = article.get("description", "")
    content = article.get("content", "")

    # content는 NewsAPI 무료 플랜에서 200자로 잘릴 수 있음
    full_text = description or content or "본문 없음"

    response = client.chat.completions.create(
        model="gpt-4o-mini",
        messages=[
            {
                "role": "system",
                "content": "뉴스 기사를 한국어로 2문장 이내로 핵심만 요약해 주세요."
            },
            {
                "role": "user",
                "content": f"제목: {title}\n\n내용: {full_text}"
            }
        ],
        max_tokens=200
    )
    return response.choices[0].message.content

def create_news_digest(keyword: str, max_articles: int = 5) -> str:
    """키워드 관련 뉴스를 수집하고 종합 다이제스트를 생성합니다."""
    articles = search_news(keyword, max_articles=max_articles)

    if not articles:
        return f"'{keyword}' 관련 최근 뉴스를 찾을 수 없습니다."

    # 각 기사 요약 수집
    summaries = []
    for i, article in enumerate(articles, 1):
        title = article.get("title", "제목 없음")
        source = article.get("source", {}).get("name", "출처 불명")
        summary = summarize_single_article(article)
        summaries.append(f"{i}. [{source}] {title}\n   → {summary}")

    summaries_text = "\n\n".join(summaries)

    # 종합 트렌드 분석
    response = client.chat.completions.create(
        model="gpt-4o-mini",
        messages=[
            {
                "role": "system",
                "content": (
                    "뉴스 요약들을 분석하여 전체적인 트렌드와 핵심 인사이트를 "
                    "3~4문장으로 종합해 주세요."
                )
            },
            {
                "role": "user",
                "content": (
                    f"'{keyword}' 관련 최신 뉴스 요약:\n\n{summaries_text}\n\n"
                    "위 뉴스들의 전반적인 트렌드를 분석해 주세요."
                )
            }
        ]
    )
    trend_analysis = response.choices[0].message.content

    result = f"## '{keyword}' 뉴스 다이제스트\n\n"
    result += f"### 개별 기사 요약\n{summaries_text}\n\n"
    result += f"### 트렌드 분석\n{trend_analysis}"
    return result

# 실행
digest = create_news_digest("생성형 AI")
print(digest)

import os
import requests
from dotenv import load_dotenv

load_dotenv()

YOUTUBE_API_KEY = os.getenv("YOUTUBE_API_KEY")
YOUTUBE_BASE_URL = "https://www.googleapis.com/youtube/v3"

def search_youtube_videos(
    query: str,
    max_results: int = 10,
    order: str = "relevance",
    published_after: str = None
) -> list[dict]:
    """YouTube에서 키워드로 영상을 검색합니다."""
    url = f"{YOUTUBE_BASE_URL}/search"
    params = {
        "part": "snippet",
        "q": query,
        "type": "video",
        "maxResults": max_results,
        "order": order,  # relevance, date, viewCount, rating
        "key": YOUTUBE_API_KEY
    }
    if published_after:
        params["publishedAfter"] = published_after  # RFC 3339 형식

    response = requests.get(url, params=params)
    response.raise_for_status()
    return response.json().get("items", [])

def get_video_details(video_ids: list[str]) -> list[dict]:
    """영상 ID 목록으로 상세 통계 정보를 가져옵니다."""
    url = f"{YOUTUBE_BASE_URL}/videos"
    params = {
        "part": "snippet,statistics,contentDetails",
        "id": ",".join(video_ids),
        "key": YOUTUBE_API_KEY
    }
    response = requests.get(url, params=params)
    response.raise_for_status()
    return response.json().get("items", [])

def get_video_comments(
    video_id: str,
    max_results: int = 20
) -> list[dict]:
    """영상의 상위 댓글을 가져옵니다."""
    url = f"{YOUTUBE_BASE_URL}/commentThreads"
    params = {
        "part": "snippet",
        "videoId": video_id,
        "maxResults": max_results,
        "order": "relevance",
        "key": YOUTUBE_API_KEY
    }
    response = requests.get(url, params=params)
    response.raise_for_status()
    return response.json().get("items", [])

def format_view_count(count: str) -> str:
    """조회수를 읽기 쉬운 형식으로 변환합니다."""
    n = int(count)
    if n >= 100_000_000:
        return f"{n / 100_000_000:.1f}억"
    elif n >= 10_000:
        return f"{n / 10_000:.1f}만"
    else:
        return f"{n:,}"

# pip install youtube-transcript-api
from youtube_transcript_api import YouTubeTranscriptApi
from openai import OpenAI

client = OpenAI()

def get_video_transcript(video_id: str, languages: list = None) -> str:
    """YouTube 영상의 자막 텍스트를 추출합니다."""
    if languages is None:
        languages = ["ko", "en"]  # 한국어 우선, 없으면 영어

    try:
        transcript_list = YouTubeTranscriptApi.get_transcript(
            video_id, languages=languages
        )
        # 타임스탬프 제거하고 텍스트만 합치기
        full_text = " ".join([t["text"] for t in transcript_list])
        return full_text
    except Exception as e:
        return f"자막을 가져올 수 없습니다: {str(e)}"

def summarize_youtube_video(video_id: str) -> dict:
    """YouTube 영상 정보와 자막을 수집하고 LLM으로 요약합니다."""
    # 영상 상세 정보 조회
    details = get_video_details([video_id])
    if not details:
        return {"error": "영상 정보를 찾을 수 없습니다."}

    video = details[0]
    snippet = video["snippet"]
    stats = video["statistics"]

    title = snippet["title"]
    description = snippet["description"][:500]  # 설명 500자 제한
    view_count = format_view_count(stats.get("viewCount", "0"))
    like_count = stats.get("likeCount", "N/A")

    # 자막 추출
    transcript = get_video_transcript(video_id)

    # 자막 길이 제한 (토큰 절약)
    if len(transcript) > 3000:
        transcript = transcript[:3000] + "... (이하 생략)"

    # LLM 요약 생성
    response = client.chat.completions.create(
        model="gpt-4o-mini",
        messages=[
            {
                "role": "system",
                "content": (
                    "YouTube 영상의 내용을 분석하여 다음을 제공해 주세요:\n"
                    "1. 핵심 내용 요약 (3~5줄)\n"
                    "2. 주요 키워드 5개\n"
                    "3. 시청 추천 대상\n"
                    "한국어로 답변해 주세요."
                )
            },
            {
                "role": "user",
                "content": (
                    f"제목: {title}\n"
                    f"설명: {description}\n\n"
                    f"자막 내용:\n{transcript}"
                )
            }
        ]
    )

    return {
        "title": title,
        "channel": snippet["channelTitle"],
        "views": view_count,
        "likes": like_count,
        "url": f"https://www.youtube.com/watch?v={video_id}",
        "summary": response.choices[0].message.content
    }

17.4 API 응답 데이터 가공과 자연어 변환 패턴

외부 API로부터 받은 JSON 구조체를 LLM이 효과적으로 처리할 수 있는 텍스트로 변환하는 것은 중요한 엔지니어링 기술입니다. 잘못된 데이터 가공은 LLM의 답변 품질을 저하시킬 수 있습니다.

데이터 가공 3원칙

원칙 1: 불필요한 기술적 필드를 제거하고 의미 있는 정보만 전달합니다. 원칙 2: 수치 데이터에 단위와 맥락을 명시합니다. 원칙 3: 중첩된 JSON은 평탄화(flatten)하여 가독성을 높입니다.

구조화 데이터 → 자연어 변환 패턴

from typing import Any

def flatten_nested_json(data: dict, prefix: str = "", sep: str = ".") -> dict:
    """중첩된 JSON을 평탄화합니다."""
    result = {}
    for key, value in data.items():
        new_key = f"{prefix}{sep}{key}" if prefix else key
        if isinstance(value, dict):
            result.update(flatten_nested_json(value, new_key, sep))
        elif isinstance(value, list) and all(isinstance(i, dict) for i in value):
            for i, item in enumerate(value):
                result.update(flatten_nested_json(item, f"{new_key}[{i}]", sep))
        else:
            result[new_key] = value
    return result

def filter_relevant_fields(
    data: dict,
    relevant_keys: list[str]
) -> dict:
    """필요한 필드만 추출합니다."""
    return {k: v for k, v in data.items() if k in relevant_keys}

def convert_to_natural_language(
    data: dict,
    field_descriptions: dict[str, str]
) -> str:
    """딕셔너리 데이터를 자연어 문장으로 변환합니다."""
    lines = []
    for field, description in field_descriptions.items():
        if field in data and data[field] is not None:
            lines.append(f"- {description}: {data[field]}")
    return "\n".join(lines)

토큰 효율을 위한 데이터 압축 전략

LLM API 호출 비용은 입력 토큰 수에 비례합니다. 대용량 API 응답을 그대로 LLM에 전달하면 비용이 급증할 수 있습니다.

def compress_api_response(
    data: Any,
    max_items: int = 5,
    max_text_length: int = 500
) -> Any:
    """API 응답을 토큰 효율적으로 압축합니다."""
    if isinstance(data, dict):
        # 불필요한 메타데이터 키 제거
        remove_keys = {"id", "url", "urlToImage", "status", "totalResults"}
        return {
            k: compress_api_response(v, max_items, max_text_length)
            for k, v in data.items()
            if k not in remove_keys
        }
    elif isinstance(data, list):
        # 리스트는 max_items 개수로 제한
        return [
            compress_api_response(item, max_items, max_text_length)
            for item in data[:max_items]
        ]
    elif isinstance(data, str) and len(data) > max_text_length:
        # 긴 텍스트는 잘라내기
        return data[:max_text_length] + "..."
    return data

def build_multi_source_context(sources: dict[str, str]) -> str:
    """여러 API 데이터를 하나의 컨텍스트 문자열로 통합합니다."""
    sections = []
    for source_name, content in sources.items():

17.5 여러 API를 조합하는 복합 연동 설계

단일 API가 아닌 여러 API를 조합하면 훨씬 풍부한 정보를 제공할 수 있습니다. 하지만 API 호출 수가 늘어날수록 레이턴시, 비용, 오류 처리 문제가 복잡해집니다.

병렬 API 호출 패턴

asyncio와 aiohttp를 사용하면 여러 API를 동시에 호출하여 응답 시간을 단축할 수 있습니다.

import asyncio
import aiohttp
from typing import Optional

async def async_get_weather(
    session: aiohttp.ClientSession,
    city: str
) -> Optional[dict]:
    """비동기 날씨 데이터 조회"""
    url = f"https://api.openweathermap.org/data/2.5/weather"
    params = {
        "q": city,
        "appid": OPENWEATHER_API_KEY,
        "units": "metric",
        "lang": "kr"
    }
    try:
        async with session.get(url, params=params) as resp:
            resp.raise_for_status()
            return await resp.json()
    except Exception as e:
        print(f"날씨 API 오류: {e}")
        return None

async def async_get_news(
    session: aiohttp.ClientSession,
    keyword: str,
    max_articles: int = 5
) -> list[dict]:
    """비동기 뉴스 데이터 조회"""
    url = f"https://newsapi.org/v2/everything"
    params = {
        "q": keyword,
        "language": "ko",
        "pageSize": max_articles,
        "sortBy": "relevancy",
        "apiKey": NEWS_API_KEY
    }
    try:
        async with session.get(url, params=params) as resp:
            resp.raise_for_status()
            data = await resp.json()
            return data.get("articles", [])
    except Exception as e:
        print(f"뉴스 API 오류: {e}")
        return []

async def async_get_youtube_videos(
    session: aiohttp.ClientSession,
    query: str,
    max_results: int = 5
) -> list[dict]:
    """비동기 YouTube 영상 검색"""
    url = f"https://www.googleapis.com/youtube/v3/search"
    params = {
        "part": "snippet",
        "q": query,
        "type": "video",
        "maxResults": max_results,
        "key": YOUTUBE_API_KEY
    }
    try:
        async with session.get(url, params=params) as resp:
            resp.raise_for_status()
            data = await resp.json()
            return data.get("items", [])
    except Exception as e:
        print(f"YouTube API 오류: {e}")
        return []

async def fetch_all_data(
    city: str,
    keyword: str
) -> tuple[Optional[dict], list, list]:
    """세 API를 병렬로 동시에 호출합니다."""
    async with aiohttp.ClientSession() as session:
        weather_task = async_get_weather(session, city)
        news_task = async_get_news(session, keyword)
        youtube_task = async_get_youtube_videos(session, keyword)

        # 세 작업을 동시에 실행
        weather, news, videos = await asyncio.gather(
            weather_task, news_task, youtube_task,
            return_exceptions=False
        )
    return weather, news, videos

오류 처리 및 폴백 전략

외부 API는 언제든지 장애가 발생할 수 있습니다. 견고한 시스템을 위해 폴백(fallback) 전략이 필요합니다.

from functools import wraps
import time

def retry_with_backoff(max_retries: int = 3, base_delay: float = 1.0):
    """지수 백오프 재시도 데코레이터"""
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            for attempt in range(max_retries):
                try:
                    return func(*args, **kwargs)
                except requests.exceptions.HTTPError as e:
                    if e.response.status_code == 429:  # Rate Limit
                        delay = base_delay * (2 ** attempt)
                        print(f"Rate limit 도달. {delay}초 후 재시도...")
                        time.sleep(delay)
                    elif attempt == max_retries - 1:
                        raise
                except requests.exceptions.RequestException as e:
                    if attempt == max_retries - 1:
                        raise
                    time.sleep(base_delay * (2 ** attempt))
            return None
        return wrapper
    return decorator

class APIOrchestrator:
    """여러 API를 조율하는 오케스트레이터 클래스"""

    def __init__(self):
        self.cache = {}  # 간단한 인메모리 캐시
        self.cache_ttl = 300  # 5분 캐시

    def _get_cached(self, key: str) -> Optional[Any]:
        if key in self.cache:
            data, timestamp = self.cache[key]
            if time.time() - timestamp < self.cache_ttl:
                return data
        return None

    def _set_cache(self, key: str, data: Any):
        self.cache[key] = (data, time.time())

    @retry_with_backoff(max_retries=3)
    def get_weather_safe(self, city: str) -> Optional[dict]:
        cached = self._get_cached(f"weather_{city}")
        if cached:
            return cached
        data = get_current_weather(city)
        self._set_cache(f"weather_{city}", data)
        return data

    @retry_with_backoff(max_retries=3)
    def get_news_safe(self, keyword: str) -> list[dict]:
        cached = self._get_cached(f"news_{keyword}")
        if cached:
            return cached
        data = search_news(keyword)
        self._set_cache(f"news_{keyword}", data)
        return data

    def get_combined_context(self, city: str, keyword: str) -> str:
        """여러 API 데이터를 통합하여 LLM 컨텍스트를 생성합니다."""
        parts = []

        # 날씨 정보 (실패해도 계속 진행)
        try:
            weather = self.get_weather_safe(city)
            if weather:
                parts.append(f"[날씨]\n{parse_weather_for_llm(weather)}")
        except Exception as e:
            parts.append(f"[날씨]\n날씨 정보를 가져올 수 없습니다. ({e})")

        # 뉴스 정보
        try:
            articles = self.get_news_safe(keyword)
            if articles:
                news_summary = "\n".join([
                    f"- {a['title']} ({a['source']['name']})"
                    for a in articles[:3]
                ])
                parts.append(f"[뉴스]\n{news_summary}")
        except Exception as e:
            parts.append(f"[뉴스]\n뉴스 정보를 가져올 수 없습니다. ({e})")

        return "\n\n".join(parts)

import os
import asyncio
import aiohttp
import requests
from openai import OpenAI
from dotenv import load_dotenv
from typing import Optional

load_dotenv()

# API 키 설정
OPENWEATHER_API_KEY = os.getenv("OPENWEATHER_API_KEY")
NEWS_API_KEY = os.getenv("NEWS_API_KEY")
YOUTUBE_API_KEY = os.getenv("YOUTUBE_API_KEY")

client = OpenAI()


# ─── 데이터 수집 함수 ────────────────────────────────────────────

def get_weather_text(city: str) -> str:
    """날씨 데이터를 텍스트로 반환합니다."""
    try:
        url = "https://api.openweathermap.org/data/2.5/weather"
        params = {
            "q": city, "appid": OPENWEATHER_API_KEY,
            "units": "metric", "lang": "kr"
        }
        resp = requests.get(url, params=params, timeout=5)
        resp.raise_for_status()
        d = resp.json()
        return (
            f"도시: {d['name']}, "
            f"기온: {d['main']['temp']}°C (체감 {d['main']['feels_like']}°C), "
            f"날씨: {d['weather'][0]['description']}, "
            f"습도: {d['main']['humidity']}%, "
            f"풍속: {d['wind']['speed']}m/s"
        )
    except Exception as e:
        return f"날씨 정보 조회 실패: {e}"


def get_news_text(keyword: str, max_articles: int = 5) -> str:
    """뉴스 데이터를 텍스트로 반환합니다."""
    try:
        url = "https://newsapi.org/v2/everything"
        params = {
            "q": keyword, "language": "ko",
            "pageSize": max_articles, "sortBy": "relevancy",
            "apiKey": NEWS_API_KEY
        }
        resp = requests.get(url, params=params, timeout=5)
        resp.raise_for_status()
        articles = resp.json().get("articles", [])
        if not articles:
            return f"'{keyword}' 관련 최근 뉴스 없음"

        lines = []
        for i, a in enumerate(articles[:max_articles], 1):
            title = a.get("title", "제목 없음")
            source = a.get("source", {}).get("name", "")
            published = a.get("publishedAt", "")[:10]
            lines.append(f"{i}. [{source}] {title} ({published})")
        return "\n".join(lines)
    except Exception as e:
        return f"뉴스 조회 실패: {e}"


def get_youtube_text(query: str, max_results: int = 5) -> str:
    """YouTube 검색 결과를 텍스트로 반환합니다."""
    try:
        url = "https://www.googleapis.com/youtube/v3/search"
        params = {
            "part": "snippet", "q": query, "type": "video",
            "maxResults": max_results, "key": YOUTUBE_API_KEY
        }
        resp = requests.get(url, params=params, timeout=5)
        resp.raise_for_status()
        items = resp.json().get("items", [])
        if not items:
            return f"'{query}' 관련 YouTube 영상 없음"

        lines = []
        for i, item in enumerate(items, 1):
            snippet = item["snippet"]
            vid_id = item["id"]["videoId"]
            title = snippet["title"]
            channel = snippet["channelTitle"]
            lines.append(
                f"{i}. {title} - {channel} "
                f"(https://youtu.be/{vid_id})"
            )
        return "\n".join(lines)
    except Exception as e:
        return f"YouTube 조회 실패: {e}"


# ─── 통합 정보 봇 클래스 ────────────────────────────────────────

class IntegratedInfoBot:
    """날씨, 뉴스, YouTube 데이터를 통합하여 답변하는 봇"""

    def __init__(self):
        self.conversation_history = []
        self.system_prompt = """당신은 실시간 데이터를 활용하는 통합 정보 어시스턴트입니다.
날씨, 뉴스, YouTube 정보를 바탕으로 사용자의 질문에 친절하고 유익하게 답변해 주세요.

답변 원칙:
1. 제공된 실제 데이터를 기반으로 답변합니다.
2. 데이터가 없는 경우 솔직하게 알립니다.
3. 필요시 추가 질문을 통해 정보를 명확히 합니다.
4. 답변은 간결하고 핵심 중심으로 작성합니다."""

    def _extract_entities(self, user_input: str) -> dict:
        """사용자 입력에서 도시명, 키워드를 추출합니다."""
        response = client.chat.completions.create(
            model="gpt-4o-mini",
            messages=[
                {
                    "role": "system",
                    "content": (
                        "사용자 메시지에서 다음을 JSON으로 추출해 주세요:\n"
                        '- city: 도시명 (없으면 "Seoul")\n'
                        "- keyword: 뉴스/YouTube 검색 키워드 (없으면 주요 주제어)\n"
                        "- needs_weather: 날씨 정보 필요 여부 (true/false)\n"
                        "- needs_news: 뉴스 정보 필요 여부 (true/false)\n"
                        "- needs_youtube: YouTube 정보 필요 여부 (true/false)\n"
                        "JSON만 출력하세요."
                    )
                },
                {"role": "user", "content": user_input}
            ],
            response_format={"type": "json_object"}
        )
        import json
        try:
            return json.loads(response.choices[0].message.content)
        except Exception:
            return {
                "city": "Seoul", "keyword": user_input,
                "needs_weather": True, "needs_news": True, "needs_youtube": True
            }

    def _gather_data(self, entities: dict) -> str:
        """엔티티 정보를 바탕으로 API 데이터를 수집합니다."""
        data_parts = []

        if entities.get("needs_weather", True):
            city = entities.get("city", "Seoul")
            weather_text = get_weather_text(city)
            data_parts.append(f"[실시간 날씨 데이터]\n{weather_text}")

        if entities.get("needs_news", True):
            keyword = entities.get("keyword", "")
            news_text = get_news_text(keyword)
            data_parts.append(f"[최신 뉴스 데이터]\n{news_text}")

        if entities.get("needs_youtube", True):
            keyword = entities.get("keyword", "")
            youtube_text = get_youtube_text(keyword)
            data_parts.append(f"[YouTube 관련 영상]\n{youtube_text}")

        return "\n\n".join(data_parts)

    def chat(self, user_input: str) -> str:
        """사용자 메시지를 받아 통합 데이터 기반 답변을 반환합니다."""
        # 1. 엔티티 추출
        entities = self._extract_entities(user_input)

        # 2. API 데이터 수집
        real_time_data = self._gather_data(entities)

        # 3. 대화 이력에 추가 (데이터 포함)
        augmented_message = (
            f"{user_input}\n\n"
            f"--- 참고할 실시간 데이터 ---\n{real_time_data}"
        )
        self.conversation_history.append({
            "role": "user",
            "content": augmented_message
        })

        # 4. LLM으로 최종 답변 생성
        messages = [
            {"role": "system", "content": self.system_prompt}
        ] + self.conversation_history

        response = client.chat.completions.create(
            model="gpt-4o-mini",
            messages=messages,
            max_tokens=1000
        )

        assistant_reply = response.choices[0].message.content

        # 대화 이력에 어시스턴트 응답 저장 (데이터 없는 원본 사용자 메시지로 교체)
        self.conversation_history[-1] = {"role": "user", "content": user_input}
        self.conversation_history.append({
            "role": "assistant",
            "content": assistant_reply
        })

        return assistant_reply

    def reset(self):
        """대화 이력을 초기화합니다."""
        self.conversation_history = []
        print("대화 이력이 초기화되었습니다.")

def main():
    bot = IntegratedInfoBot()

    print("=" * 60)
    print("통합 정보 봇에 오신 것을 환영합니다!")
    print("날씨, 뉴스, YouTube 정보를 실시간으로 제공합니다.")
    print("종료하려면 'quit' 또는 'exit'를 입력하세요.")
    print("대화 초기화는 'reset'을 입력하세요.")
    print("=" * 60)

    while True:
        user_input = input("\n질문: ").strip()

        if not user_input:
            continue
        if user_input.lower() in ["quit", "exit"]:
            print("봇을 종료합니다. 감사합니다!")
            break
        if user_input.lower() == "reset":
            bot.reset()
            continue

        print("\n답변 생성 중...")
        response = bot.chat(user_input)
        print(f"\n봇: {response}")


if __name__ == "__main__":
    main()
