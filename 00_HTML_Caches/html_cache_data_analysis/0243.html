import matplotlib.pyplot as plt
import numpy as np
dice = np.random.choice(6, 10)

plt.hist(dice, bins =6)
plt.show()
 
import matplotlib.pyplot as plt
import random
dice = []
for i in range(10) :
    dice.append(random.randint(1,6))
plt.hist(dice, bins = 6)
plt.show()
