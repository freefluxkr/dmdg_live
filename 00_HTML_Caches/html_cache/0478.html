import matplotlib.pyplot as plt

s_mon = [0, 0, 0]
for row in data:
    if row[-1] == '전표매입':
        mon, payment = int(row[0].split('-')[1]), int(row[-3])
        idx = mon - 10
        s_mon[idx] += payment

plt.rc('font', family='Malgun Gothic')
plt.title('10~12월 지출현황')
plt.bar(['10월', '11월', '12월'], s_mon, color='royalblue')
plt.show()
