print('미세먼지 저감 조치에 따른 차량 2부제를 시행합니다!')
date = int(input('오늘은 며칠입니까? '))
if date % 2 == 1:
    print('홀수번호 차량만 통행 가능합니다')
    print('짝수번호 차주는 오늘 대중교통을 이용하세요.')
else:
    print('짝수번호 차량만 통행 가능합니다.')
    print('홀수번호 차주는 오늘 대중교통을 이용하세요.')
