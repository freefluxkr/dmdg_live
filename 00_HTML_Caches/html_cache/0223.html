import random
print('구구단을 외자! 구구단을 외자!')
for i in range(10):
    a, b = random.randint(2, 9), random.randint(1, 9)
    answer = input(str(a) + 'x' + str(b) + '? ')
    answer = int(answer)
    if answer != a * b:
        print('틀렸습니다! 5번 반복해 읽어 봅시다.')
        for j in range(5):
            print(a, 'x', b, '?', a * b)
