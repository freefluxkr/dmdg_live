for row in data:
    for i in range(1, 32):
        if i <= 2:      # 인덱스가 2 이하인 경우 정수형으로 변환
            row[i] = int(row[i])
        else:           # 인덱스가 2 초과인 경우 실수형으로 변환
            row[i] = float(row[i])
