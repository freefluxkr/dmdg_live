앞에서 작성했던 코드를 조금 변형해서 연령대별 남성 인구에서 여성 인구를 뺀 값을 result라는 리스트에 추가합니다.

import csv
f = open('gender.csv')
data = csv.reader(f)
result = []
name = input('궁금한 동네를 입력해주세요:')
for row in data :
    if name in row[0] :
        for i in range(3,104) :
            result.append(int(row[i]) - int(row[i+103]))
        break
